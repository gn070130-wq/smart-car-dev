//---------------------------------------------------------------------------------------------------
// AHRS algorithm update

void MahonyAHRSupdate(float gx, float gy, float gz, float ax, float ay, float az, float mx, float my, float mz)
{
    float recipNorm;
    float q0q0, q0q1, q0q2, q0q3, q1q1, q1q2, q1q3, q2q2, q2q3, q3q3;
    float hx, hy, bx, bz;
    float halfvx, halfvy, halfvz, halfwx, halfwy, halfwz;
    float halfex, halfey, halfez;
    float qa, qb, qc;
    gx *= AtR;
    gy *= AtR;
    gz *= AtR;
    // Use IMU algorithm if magnetometer measurement invalid (avoids NaN in magnetometer normalisation
    if((mx == 0.0f) && (my == 0.0f) && (mz == 0.0f))
    {
        MahonyAHRSupdateIMU(gx, gy, gz, ax, ay, az);
        return;
    }

    // Compute feedback only if accelerometer measurement valid (avoids NaN in accelerometer normalisation)
    if(!((ax == 0.0f) && (ay == 0.0f) && (az == 0.0f)))
    {

            // Normalise accelerometer measurement
            recipNorm = invSqrt(ax * ax + ay * ay + az * az);
            ax *= recipNorm;
            ay *= recipNorm;
            az *= recipNorm;

            // Normalise magnetometer measurement
            recipNorm = invSqrt(mx * mx + my * my + mz * mz);
            mx *= recipNorm;
            my *= recipNorm;
            mz *= recipNorm;

            // Auxiliary variables to avoid repeated arithmetic
            q0q0 = q0 * q0;
            q0q1 = q0 * q1;
            q0q2 = q0 * q2;
            q0q3 = q0 * q3;
            q1q1 = q1 * q1;
            q1q2 = q1 * q2;
            q1q3 = q1 * q3;
            q2q2 = q2 * q2;
            q2q3 = q2 * q3;
            q3q3 = q3 * q3;

            // Reference direction of Earth's magnetic field
            hx = 2.0f * (mx * (0.5f - q2q2 - q3q3) + my * (q1q2 - q0q3) + mz * (q1q3 + q0q2));
            hy = 2.0f * (mx * (q1q2 + q0q3) + my * (0.5f - q1q1 - q3q3) + mz * (q2q3 - q0q1));
            bx = sqrt(hx * hx + hy * hy);
            bz = 2.0f * (mx * (q1q3 - q0q2) + my * (q2q3 + q0q1) + mz * (0.5f - q1q1 - q2q2));

            // Estimated direction of gravity and magnetic field
            halfvx = q1q3 - q0q2;
            halfvy = q0q1 + q2q3;
            halfvz = q0q0 - 0.5f + q3q3;
            halfwx = bx * (0.5f - q2q2 - q3q3) + bz * (q1q3 - q0q2);
            halfwy = bx * (q1q2 - q0q3) + bz * (q0q1 + q2q3);
            halfwz = bx * (q0q2 + q1q3) + bz * (0.5f - q1q1 - q2q2);

            // Error is sum of cross product between estimated direction and measured direction of field vectors
            halfex = (ay * halfvz - az * halfvy) + (my * halfwz - mz * halfwy);
            halfey = (az * halfvx - ax * halfvz) + (mz * halfwx - mx * halfwz);
            halfez = (ax * halfvy - ay * halfvx) + (mx * halfwy - my * halfwx);

            // Compute and apply integral feedback if enabled
            if(twoKi > 0.0f)
            {
                integralFBx += twoKi * halfex * (1.0f / sampleFreq);    // integral error scaled by Ki
                integralFBy += twoKi * halfey * (1.0f / sampleFreq);
                integralFBz += twoKi * halfez * (1.0f / sampleFreq);
                gx += integralFBx;  // apply integral feedback
                gy += integralFBy;
                gz += integralFBz;
            }
            else
            {
                integralFBx = 0.0f; // prevent integral windup
                integralFBy = 0.0f;
                integralFBz = 0.0f;
            }

            // Apply proportional feedback
            gx += twoKp * halfex;
            gy += twoKp * halfey;
            gz += twoKp * halfez;
    }

    // Integrate rate of change of quaternion
    gx *= (0.5f * (1.0f / sampleFreq));     // pre-multiply common factors
    gy *= (0.5f * (1.0f / sampleFreq));
    gz *= (0.5f * (1.0f / sampleFreq));
    qa = q0;
    qb = q1;
    qc = q2;
    q0 += (-qb * gx - qc * gy - q3 * gz);
    q1 += (qa * gx + qc * gz - q3 * gy);
    q2 += (qa * gy - qb * gz + q3 * gx);
    q3 += (qa * gz + qb * gy - qc * gx);

    // Normalise quaternion
    recipNorm = invSqrt(q0 * q0 + q1 * q1 + q2 * q2 + q3 * q3);
    q0 *= recipNorm;
    q1 *= recipNorm;
    q2 *= recipNorm;
    q3 *= recipNorm;

        Angle_Z_Quaternions = atan2(2*q1*q2+2*q0*q3,-q3*q3-q2*q2+q0*q0+q1*q1)* RtA;//四元数获取偏航角

        if(Angle_Z_Quaternions < 0)
            Angle_Z_Quaternions = Angle_Z_Quaternions + 360;
        if(Angle_Z_Quaternions - angle_Z_Quaternions > 1 || Angle_Z_Quaternions - angle_Z_Quaternions  < -1)
            angle_Z_Quaternions = Angle_Z_Quaternions;
        else
            Angle_Z_Quaternions = angle_Z_Quaternions;
}

    {
        // Normalise accelerometer measurement
        recipNorm = invSqrt(ax * ax + ay * ay + az * az);

        ay *= recipNorm;

        halfvy = q0 * q1 + q2 * q3;

//---------------------------------------------------------------------------------------------------
// Fast inverse square-root
// See: http://en.wikipedia.org/wiki/Fast_inverse_square_root
//快速开平方，只是他返回的时倒数，如，invSqrt（5）  他返回的时 1/根号5
float invSqrt(float x)
{
    float halfx = 0.5f * x;
    float y = x;
    long i = *(long*)&y;
    i = 0x5f3759df - (i>>1);
    y = *(float*)&i;
    y = y * (1.5f - (halfx * y * y));
    return y;
}

msdf_LowPassFilter2p_t mag_filter[3];
//设置iir 二阶滤波器参数
void msdf_set_cutoff_frequency(float sample_freq, float cutoff_freq, msdf_LowPassFilter2p_t *lp)
{
        lp->cutoff_freq = cutoff_freq;

        if (lp->cutoff_freq <= 0.0f) {
                return;
        }

        float fr = sample_freq / lp->cutoff_freq;
        float ohm = tanf(PI / fr);
        float c = 1.0f + 2.0f * cosf(PI / 4.0f) * ohm + ohm * ohm;
        lp->b0 = ohm * ohm / c;
        lp->b1 = 2.0f * lp->b0;
        lp->b2 = lp->b0;
        lp->a1 = 2.0f * (ohm * ohm - 1.0f) / c;
        lp->a2 = (1.0f - 2.0f * cosf(PI / 4.0f) * ohm + ohm * ohm) / c;
}

//二阶滤波器运行
float msdf_lowpass2p_apply(float sample, msdf_LowPassFilter2p_t *lp)
{
        if (lp->cutoff_freq <= 0.0f) {
                return sample;
        }

        float delay_element_0 = sample - lp->delay_element_1 * lp->a1 - lp->delay_element_2 * lp->a2;

        if (!isfinite(delay_element_0))
        {
                delay_element_0 = sample;
        }

        float output = delay_element_0 * lp->b0 + lp->delay_element_1 * lp->b1 + lp->delay_element_2 * lp->b2;

        lp->delay_element_2 = lp->delay_element_1;
        lp->delay_element_1 = delay_element_0;

        return output;
}

            scale_y = (double)display_high / height;

            scale = fmin(scale_x, scale_y);

            // 注意：当 X 轴被镜像反转后，居中的 offset 逻辑通常依然适用，
            // 因为我们是整体翻转了坐标空间。
            if (scale == scale_x)
            {
                y_offset = ((double)display_high - height * scale) / 2.0;
            }
            else
            {
                x_offset = ((double)display_width - width * scale) / 2.0;
            }

            for (int i = 0; i < num-1; i++)
            {
                // 【修改点】X 轴镜像：加上 "display_width - " 实现左右翻转
                // 原公式：(TempLat - min) * scale + offset
                // 新公式：总宽 - (原公式)
                double temp_x = (TempLat[i+1] - min_x) * scale + x_offset;
                screen_point_data[i].x = (double)display_width - temp_x;

                // Y 轴保持不变 (保持您之前的逻辑，无论是反转还是直连)
                // 如果之前 Y 轴是反转的 (上北下南)，这里保持不动即可
                screen_point_data[i].y = (double)display_high - ((TempLon[i+1] - min_y) * scale + y_offset);
            }
        }
    }
}

        }

        }
     }
     else if(angle == 0)              //x轴的正半轴上

        {
            return 0;
        }
        if(X_err < 0)                 //x轴的负半轴上

            return 180;

void gps_turn(void) {
    // 1. 获取起点和终点经纬度
    double lat1 = TempLat[0]; // 假设这里存的是真实的纬度(度)
    double lon1 = TempLon[0]; // 假设这里存的是真实的经度(度)
    double lat2 = gnss.latitude;
    double lon2 = gnss.longitude;

    // 2. 定义地球半径和常量
    const double EARTH_RADIUS = 6371000.0; // 米
    const double DEG_TO_RAD = PI / 180.0;

    // 3. 计算经纬度差值 (转为弧度)
    double dLat = (lat2 - lat1) * DEG_TO_RAD;
    double dLon = (lon2 - lon1) * DEG_TO_RAD;

    // 计算平均纬度 (用于经度修正)
    double avgLat = (lat1 + lat2) / 2.0 * DEG_TO_RAD;

    // 4. 【核心修正】直接计算平面坐标 (ENU: East-North-Up)
    // 北向位移 (X) = 纬度差 * 地球半径
    double north_meters = dLat * EARTH_RADIUS;

    // 东向位移 (Y) = 经度差 * 地球半径 * cos(平均纬度)
    // 这一步自动解决了“高纬度经度失真”的问题
    double east_meters = dLon * EARTH_RADIUS * cos(avgLat);

    // 5. 存入结果 (明确这是平面坐标，单位是米)
    // 建议修改变量名或加注释，避免混淆
    TempLat[num] = north_meters;  // X 轴：北向距离
    TempLon[num] = east_meters;   // Y 轴：东向距离

    // 此时，如果您再用之前的 get_fang_wei_jiao 计算：
    // angle = atan2(TempLon[next]-TempLon[now], TempLat[next]-TempLat[now])
    // 结果将会非常准确，且不会再出现 X 为负数的情况（如果是向北走的话）。
}

//**26.3.9优化获取方位角函数**//
double get_fang_wei_jiao(double X_now, double Y_now, double X_next, double Y_next) {
    double dx = X_next - X_now; // 北向位移
    double dy = Y_next - Y_now; // 东向位移

    // 使用 atan2 替代 atan，避免除以零和精度损失
    // atan2(y, x) 返回 [-PI, PI]
    // 在 X=北，Y=东 的坐标系下：
    // dx>0, dy=0 (正北) -> atan2(0, pos) = 0
    // dx=0, dy>0 (正东) -> atan2(pos, 0) = PI/2 (90度)
    // 这正好符合地理方位角的定义 (0北，90东，顺时针增加)
    double angle_rad = atan2(dy, dx);

    double angle_deg = angle_rad * 180.0 / PI;

    // 将 [-180, 180] 转换为 [0, 360]
    if (angle_deg < 0) {
        angle_deg += 360.0;
    }

    return angle_deg;
}

    // --- 第一步：处理经度页 (Page 1) ---
    flash_erase_page(0, 1);             // 擦除扇区第一页
    flash_buffer_clear();               // 清空缓存

    // [重要] 写入帧头信息
    // 假设 flash_union_buffer[0] 存帧头，flash_union_buffer[1] 存有效点数 num
    // 请根据实际 temp_max 调整，确保不越界。如果 temp_max 很小，可能需要压缩存储或单独定义结构体
    if (temp_max < 2) {
        // 错误处理：缓冲区太小，无法存放帧头
        return;
    }

    flash_union_buffer[0].int32_type = GPS_DATA_MAGIC; // 索引0：帧头
    flash_union_buffer[1].int32_type = num;            // 索引1：当前有效点数

    // 写入经度数据 (从索引2开始存放实际数据，防止覆盖帧头)
    // 注意：这里需要确保 temp_max + 2 不超过单页缓存容量
    for(int i = 0; i < num && (i + 2) < temp_max; i++)
    {
        x1 = Disassembly_GPS(TempLon[i]);
        // 特殊处理：如果转换后为0，且你想区分"0度"和"无效"，可以在此处 +1 或其他偏移，读取时再减回去
        // 这里假设 0 是合法的经纬度值，依靠帧头来判断有效性
        flash_union_buffer[i + 2].int32_type = x1;
    }

    // 填充剩余部分为无效标记 (可选，防止读取旧数据)
    for(int i = num + 2; i < temp_max; i++) {
        flash_union_buffer[i].int32_type = -1; // 或者 0xFFFFFFFF
    }

    flash_write_page_from_buffer(0, 1);       // 写入经度页

    // --- 第二步：处理纬度页 (Page 2) ---
    flash_erase_page(0, 2);             // 擦除扇区第二页
    flash_buffer_clear();               // 清空缓存

    // [重要] 纬度页也写入同样的帧头，用于双重校验或独立校验
    // 为了简化，纬度页也可以只存数据，依靠经度页的帧头。
    // 但为了安全，建议在纬度页也存一份 num，或者存一个校验和。
    // 这里采用简单策略：纬度页索引0也存帧头，表示纬度页也是新的
    flash_union_buffer[0].int32_type = GPS_DATA_MAGIC;
    flash_union_buffer[1].int32_type = num;

    for(int i = 0; i < num && (i + 2) < temp_max; i++)
    {
        x2 = Disassembly_GPS(TempLat[i]);
        flash_union_buffer[i + 2].int32_type = x2;
    }

    // 填充剩余部分
    for(int i = num + 2; i < temp_max; i++) {
        flash_union_buffer[i].int32_type = -1;
    }

    flash_write_page_from_buffer(0, 2);       // 写入纬度页
}

{

        stored_num_lon = flash_union_buffer[1].int32_type;

        // 校验帧头 和 数量范围
        if (magic_lon == GPS_DATA_MAGIC && stored_num_lon > 0 && stored_num_lon < temp_max - 2)
        {
            // 校验通过，开始读取数据
            // 注意：数据是从索引 2 开始存的
            for(int i = 0; i < stored_num_lon; i++)
            {
                y1 = flash_union_buffer[i + 2].int32_type;

                // 额外校验：如果是 -1 (0xFFFFFFFF)，说明是填充的无效数据，停止读取
                if (y1 == -1 || y1 == 0xFFFFFFFF) break;

                TempLon[i] = Package_GPS(y1);
            }

            // 暂时记录从经度页读到的数量

    if (num == 0) return;

    if (flash_check(0, 2))
    {
        flash_read_page_to_buffer(0, 2);

        magic_lat = flash_union_buffer[0].int32_type;
        stored_num_lat = flash_union_buffer[1].int32_type;

        // 校验纬度页：帧头必须对，且记录的数量必须与经度页一致（保证成对）
        if (magic_lat == GPS_DATA_MAGIC && stored_num_lat == num)
        {
            for(int i = 0; i < num; i++)
            {
                y2 = flash_union_buffer[i + 2].int32_type;
                if (y2 == -1 || y2 == 0xFFFFFFFF) break;

                TempLat[i] = Package_GPS(y2);
            }
            // num 保持不变，因为已经确认经纬度成对且完整
        } else {

            // for(int i=0; i<temp_max; i++) TempLon[i] = 0;
        }
        flash_buffer_clear();
    } else {
        // 有经度没纬度，数据不完整
        num = 0;
    }
}

// --- 测速滤波配置 ---26.3.21
#define SPEED_FILTER_ALPHA  0.03f       // 采样率低了，稍微调大一点 Alpha 以减少延迟 (原 0.15)
#define MAX_ENCODER_PULSE   800         // 20ms 周期，阈值放大
#define MAX_SPEED_DELTA     0.02f        // 20ms 允许的变化量也可以稍微放宽
#define SPEED_COEFFICIENT   0.036825f   // 【核心】0.1473 / 4.0

// --- 滤波器状态变量 (右轮) ---
float g_filtered_R_count = 0.0f;        // 滤波后的脉冲计数
float g_last_R_speed     = 0.0f;        // 上一次的速度值
int   g_filter_R_init    = 0;           // 滤波器初始化标志

// --- 滤波器状态变量 (左轮) ---
float g_filtered_L_count = 0.0f;        // 滤波后的脉冲计数
float g_last_L_speed     = 0.0f;        // 上一次的速度值
int   g_filter_L_init    = 0;           // 滤波器初始化标志

//-------------------------------------------------------------------------------------------------------------------
// 函数简介     获取右轮速度 (优化版：带滤波、限幅、防野值)
// 参数说明     void
// 返回参数     void
// 备注信息     在 20ms 定时器中断中调用//26.3.21
//-------------------------------------------------------------------------------------------------------------------
void get_car_Rspeed(void)
{
    float raw_count = 0.0f;
    float current_speed = 0.0f;

    // 1. 读取编码器原始计数值 (注意符号，根据您的硬件方向决定是否需要负号)
    raw_count = (float)encoder_get_count(TIM6_ENCODER);

    // 如果您的编码器方向反了，这里加负号；如果方向正确，去掉负号
    // 原代码是: co = -encoder_get_count(...); 这里保持一致
    raw_count = -raw_count;

    // 2. 【重要】野值截断保护
    // 如果 5ms 内脉冲数超过物理极限(如>180)，认为是干扰，强制限制在阈值
    if(raw_count > MAX_ENCODER_PULSE) {
        raw_count = MAX_ENCODER_PULSE;
    } else if(raw_count < -MAX_ENCODER_PULSE) {
        raw_count = -MAX_ENCODER_PULSE;
    }

    // 3. 【核心】一阶低通滤波 (替代原来的滑动平均)
    // 公式：New_Filter = Old_Filter * (1-Alpha) + New_Raw * Alpha
    if(g_filter_R_init == 0) {
        g_filtered_R_count = raw_count; // 第一次直接赋值
        g_filter_R_init = 1;
    } else {
        g_filtered_R_count = g_filtered_R_count * (1.0f - SPEED_FILTER_ALPHA) + raw_count * SPEED_FILTER_ALPHA;
    }

    // 4. 清零计数器
    encoder_clear_count(TIM6_ENCODER);

    // 5. 计算速度 (m/s)
    current_speed = g_filtered_R_count * SPEED_COEFFICIENT;

    // 6. 【增强】速度变化率限制 (防跳变)
    // 如果当前速度与上次速度差值过大，强制限制变化幅度
    float delta = current_speed - g_last_R_speed;
    if(delta > MAX_SPEED_DELTA) {
        current_speed = g_last_R_speed + MAX_SPEED_DELTA;
    } else if(delta < -MAX_SPEED_DELTA) {
        current_speed = g_last_R_speed - MAX_SPEED_DELTA;
    }

    // 更新历史速度
    g_last_R_speed = current_speed;

    // 7. 赋值给系统全局变量
    ins.Rspeed = current_speed;
}

//-------------------------------------------------------------------------------------------------------------------
// 函数简介     位置式PID
// 参数说明     void
// 返回参数     void
// 使用示例
// 备注信息    这个函数只是一个模板，使用需要 Pid_Pos 来定义一个结构体，只要是所有的使用位置式PID都可以使用这个函数
//---------------------------------------------------------------------------------------------------------------
void Pid_Position(Pid_Pos *pid)
{
    pid->Err = pid->Set - pid->Actual;

    pid->increment += pid->Err;

    pid->increment = limit_value(pid->increment,pid->Limit_increment,(-pid->Limit_increment));

    pid->proportion =  pid->P * pid->Err;

    pid->integration = pid->I * pid->increment;

    pid->differential = (pid->Err - pid->Err_Next) * pid->D;

    pid->Err_Next= pid->Err;

    pid->Out = pid->proportion  + pid->integration + pid->differential + pid->KK * pid->Set;
}
//-------------------------------------------------------------------------------------------------------------------
// 函数简介     增量式PID
// 参数说明     void
// 返回参数     void
// 使用示例
// 备注信息    这个函数只是一个模板，使用需要 Pid_Inc 来定义一个结构体，只要是所有的使用增量式PID都可以使用这个函数
//---------------------------------------------------------------------------------------------------------------
void Pid_Increse(Pid_Inc *pid)
{
    pid->Err = pid->Set - pid->Actual;

    pid->proportion = pid->P * (pid->Err - pid->Err_Next);

    pid->integration = pid->I * pid->Err;

    pid->differential = pid->D * (pid->Err - 2*pid->Err_Next + pid->Err_Last);

    pid->increment = pid->proportion + pid->integration + pid->differential;

    pid->Out += pid->increment;

    pid->Out = limit_value(pid->Out , pid->Limit_increment , -pid->Limit_increment);

    pid->Err_Last = pid->Err_Next;

    pid->Err_Next = pid->Err;
}

//-------------------------------------------------------------------------------------------------------------------
// 函数简介     限幅函数
// 参数说明     void
// 返回参数     void
// 使用示例
// 备注信息
//---------------------------------------------------------------------------------------------------------------
float limit_value(float value,float positive_value,float negative_value)
{
    if(value>positive_value)
    {
        value=positive_value;
    }
    if(value<negative_value)
    {
        value=negative_value;
    }
    return value;
}

void Writ_IMU_flash(void)
{
    int32 x1 = 0, x2 = 0;

    // --- 第一步：处理距离页 (Page 5) ---
    flash_erase_page(0, 5);             // 擦除扇区第 5 页
    flash_buffer_clear();               // 清空缓存

    // 边界检查：确保 imu_num 不超过 Flash 单页缓存容量 (temp_max)
    int32 write_count = (imu_num > temp_max) ? temp_max : imu_num;

    // 写入距离数据 (从索引 0 开始存放)
    for(int i = 0; i < write_count; i++)
    {
        x1 = Disassembly_GPS(Tempdistance[i]);
        flash_union_buffer[i].int32_type = x1;
    }

    // 填充剩余部分为无效标记
    for(int i = write_count; i < temp_max; i++) {
        flash_union_buffer[i].int32_type = -1;
    }

    flash_write_page_from_buffer(0, 5);       // 写入距离页 (Page 5)

    // --- 第二步：处理角度页 (Page 6) ---
    flash_erase_page(0, 6);             // 擦除扇区第 6 页
    flash_buffer_clear();               // 清空缓存

    // 写入角度数据 (从索引 0 开始存放)
    for(int i = 0; i < write_count; i++)
    {
        x2 = Disassembly_GPS(Tempangle[i]);
        flash_union_buffer[i].int32_type = x2;
    }

    // 填充剩余部分
    for(int i = write_count; i < temp_max; i++) {
        flash_union_buffer[i].int32_type = -1;
    }

    flash_write_page_from_buffer(0, 6);       // 写入角度页 (Page 6)
}

//-------------------------------------------------------------------------------------------------------------------
// 函数简介     从 Flash 读取 IMU/距离角度数据 (第 5、6 页)
// 修改说明     对应 Writ_IMU_flash，通过校验数据完整性恢复 imu_num
//-------------------------------------------------------------------------------------------------------------------
void Read_IMU_flash(void)
{
    int32 y1, y2;
    int32 count_dist = 0, count_angle = 0;

    // 初始化：默认无数据
    imu_num = 0;

    // --- 检查距离页 (Page 5) ---
    if (flash_check(0, 5))
    {
        flash_read_page_to_buffer(0, 5);

        // 统计有效数据个数 (从索引 0 开始)
        for(int i = 0; i < temp_max; i++)
        {
            y1 = flash_union_buffer[i].int32_type;

            // 遇到无效标记停止
            if (y1 == -1 || y1 == 0xFFFFFFFF)
            {
                break;
            }

            // 还原数据
            Tempdistance[i] = Package_GPS(y1);
            count_dist++;
        }

        // 暂时记录数量
        imu_num = count_dist;
        flash_buffer_clear();
    }

    // 如果距离页没数据，直接返回
    if (imu_num == 0) return;

    // --- 检查角度页 (Page 6) ---
    if (flash_check(0, 6))
    {
        flash_read_page_to_buffer(0, 6);

        // 统计角度页有效数据
        for(int i = 0; i < temp_max; i++)
        {
            y2 = flash_union_buffer[i].int32_type;
            if (y2 == -1 || y2 == 0xFFFFFFFF) break;

            Tempangle[i] = Package_GPS(y2);
            count_angle++;
        }

        // 校验：距离和角度的数量必须一致
        if (count_angle == imu_num)
        {
            // 数据完整，imu_num 保持不变
        } else {
            // 数量不匹配，数据视为无效
            imu_num = 0;
        }
        flash_buffer_clear();
    } else {
        // 有距离没角度，数据不完整
        imu_num = 0;
    }
}

void Writ_IMU_flashkm3(void)
{
    int32 x1 = 0, x2 = 0;

    // --- 第一步：处理距离页 (Page 9) ---
    flash_erase_page(0, 9);             // 擦除扇区第 9 页
    flash_buffer_clear();               // 清空缓存

    // 边界检查：确保 imu_num 不超过 Flash 单页缓存容量 (temp_max)
    int32 write_count = (imu_num > temp_max) ? temp_max : imu_num;

    // 写入距离数据 (从索引 0 开始存放)
    for(int i = 0; i < write_count; i++)
    {
        x1 = Disassembly_GPS(Tempdistance[i]);
        flash_union_buffer[i].int32_type = x1;
    }

    // 填充剩余部分为无效标记
    for(int i = write_count; i < temp_max; i++) {
        flash_union_buffer[i].int32_type = -1;
    }

    flash_write_page_from_buffer(0, 9);       // 写入距离页 (Page 9)

    // --- 第二步：处理角度页 (Page 10) ---
    flash_erase_page(0, 10);             // 擦除扇区第 10 页
    flash_buffer_clear();               // 清空缓存

    // 写入角度数据 (从索引 0 开始存放)
    for(int i = 0; i < write_count; i++)
    {
        x2 = Disassembly_GPS(Tempangle[i]);
        flash_union_buffer[i].int32_type = x2;
    }

    // 填充剩余部分
    for(int i = write_count; i < temp_max; i++) {
        flash_union_buffer[i].int32_type = -1;
    }

    flash_write_page_from_buffer(0, 10);       // 写入角度页 (Page 10)
}

//数组原地逆序函数（支持 int32 类型）
void reverse_array(int32 arr[], int len)
{
    int left = 0;
    int right = len - 1;
    int32 temp;
    while (left < right) {
        // 交换左右两端的数据
        temp = arr[left];
        arr[left] = arr[right];
        arr[right] = temp;
        // 指针向中间靠拢
        left++;
        right--;
    }
}

void Read_IMU_flashkm3(void)
{
    int32 y1, y2;
    int32 count_dist = 0, count_angle = 0;

    // 初始化：默认无数据
    imu_num = 0;

    // --- 检查距离页 (Page 9) ---
    if (flash_check(0, 9))
    {
        flash_read_page_to_buffer(0, 9);

        // 统计有效数据个数 (从索引 0 开始)
        for(int i = 0; i < temp_max; i++)
        {
            y1 = flash_union_buffer[i].int32_type;

            // 遇到无效标记停止
            if (y1 == -1 || y1 == 0xFFFFFFFF)
            {
                break;
            }

            // 还原数据
            Tempdistance[i] = Package_GPS(y1);
            if(i == 0) Tempdistance[i] = Package_GPS(y1) + 40;
            count_dist++;
        }

        // 暂时记录数量
        imu_num = count_dist;
        flash_buffer_clear();
    }

    // 如果距离页没数据，直接返回
    if (imu_num == 0) return;

    // --- 检查角度页 (Page 10) ---
    if (flash_check(0, 10))
    {
        flash_read_page_to_buffer(0, 10);

        // 统计角度页有效数据
        for(int i = 0; i < temp_max; i++)
        {
            y2 = flash_union_buffer[i].int32_type;
            if (y2 == -1 || y2 == 0xFFFFFFFF) break;

           Tempangle[i] = Package_GPS(y2);

           int16_t angle_x10 = (int16_t)(Tempangle[i] * 10);

           // 【核心技巧】将角度整体偏移 +45度(即+450)，把跨零点问题转化为普通区间问题
           angle_x10 += 450;
           if (angle_x10 >= 3600) {
               angle_x10 -= 3600; // 如果偏移后超过3600，减去一圈拉回正常范围
           }

           // 使用纯整数 if-else 进行区间映射
           int16_t result_angle;
           if (angle_x10 < 900) {      // 对应原角度的 335° ~ 45°
               result_angle = 0;
           }
           else if (angle_x10 < 1800) { // 对应原角度的 45° ~ 135°

           Tempangle[i] = (float)result_angle;

           Tempangle[i] = fmod(Tempangle[i] + 180 , 360);
            count_angle++;
        }

        // 校验：距离和角度的数量必须一致
        if (count_angle == imu_num)
        {
            // 数据完整，执行逆序读取（反转数组）
            reverse_array(Tempdistance,imu_num);
            reverse_array(Tempangle, imu_num);
        } else {
            // 数量不匹配，数据视为无效
            imu_num = 0;
        }
        flash_buffer_clear();
    } else {
        // 有距离没角度，数据不完整
        imu_num = 0;
    }
}

void ReadColor(unsigned int x,unsigned int y,COLOR_RGB *Rgb)
{
    unsigned short C16;

    C16 = GUI_ReadBit16Point(x,y);

    Rgb->red   =     (unsigned char)((C16&0xf800)>>8);
    Rgb->green =     (unsigned char)((C16&0x07e0)>>3);
    Rgb->blue  =     (unsigned char)((C16&0x001f)<<3);
}

//RGB转HSL
 void RGBtoHSL(const COLOR_RGB *Rgb, COLOR_HSL *Hsl)
{
    int h,s,l,maxVal,minVal,difVal;
    int r  = Rgb->red;
    int g  = Rgb->green;
  int b  = Rgb->blue;

    maxVal = max3v(r, g, b);
    minVal = min3v(r, g, b);

    difVal = maxVal-minVal;

    //计算亮度
    l = (maxVal+minVal)*240/255/2;

    if(maxVal == minVal)//若r=g=b
    {
        h = 0;
        s = 0;
    }
    else
    {
        //计算色调
        if(maxVal==r)
        {
            if(g>=b)
                h = 40*(g-b)/(difVal);
            else
                h = 40*(g-b)/(difVal) + 240;
        }
        else if(maxVal==g)
            h = 40*(b-r)/(difVal) + 80;
        else if(maxVal==b)
            h = 40*(r-g)/(difVal) + 160;
        //计算饱和度
        if(l == 0)
            s = 0;
        else if(l<=120)
            s = (difVal)*240/(maxVal+minVal);
        else
            s = (difVal)*240/(480 - (maxVal+minVal));
    }
    Hsl->hue =        (unsigned char)(((h>240)? 240 : ((h<0)?0:h)));
    Hsl->saturation = (unsigned char)(((s>240)? 240 : ((s<0)?0:s)));
    Hsl->luminance =  (unsigned char)(((l>240)? 240 : ((l<0)?0:l)));
}

//匹配颜色
static int ColorMatch(const COLOR_HSL *Hsl,const TARGET_CONDI *Condition)
{
    if(
        Hsl->hue        >=  Condition->H_MIN &&
        Hsl->hue        <=  Condition->H_MAX &&
        Hsl->saturation >=  Condition->S_MIN &&
        Hsl->saturation <=   Condition->S_MAX &&
        Hsl->luminance  >=  Condition->L_MIN &&
        Hsl->luminance  <=   Condition->L_MAX
    )
        return 1;
    else
        return 0;
}

//搜索腐蚀中心
static int SearchCentre(unsigned int *x,unsigned int *y,const TARGET_CONDI *Condition,const SEARCH_AREA *Area)
{
    unsigned int SpaceX,SpaceY,i,j,k,FailCount=0;
    COLOR_RGB Rgb;
    COLOR_HSL Hsl;

    SpaceX = Condition->WIDTH_MIN/3;
    SpaceY = Condition->HIGHT_MIN/3;

    for(i=Area->Y_Start;i<Area->Y_End;i+=SpaceY)
    {
        for(j=Area->X_Start;j<Area->X_End;j+=SpaceX)
        {
            FailCount=0;
            for(k=0;k<SpaceX+SpaceY;k++)
            {
                if(k<SpaceX)
                    ReadColor(j+k,i+SpaceY/2,&Rgb);
                else
                    ReadColor(j+SpaceX/2,i+(k-SpaceX),&Rgb);
                RGBtoHSL(&Rgb,&Hsl);

                if(!ColorMatch(&Hsl,Condition))
                    FailCount++;
                if(FailCount>((SpaceX+SpaceY)>>ALLOW_FAIL_PER))
                    break;
            }
            if(k==SpaceX+SpaceY)
            {
                *x = j+SpaceX/2;
                *y = i+SpaceY/2;
                return 1;
            }
        }
    }
    return 0;
}

//从腐蚀中心向外腐蚀，得到新的腐蚀中心
static int Corrode(unsigned int oldx,unsigned int oldy,const TARGET_CONDI *Condition,RESULT *Resu)
{
    unsigned int Xmin,Xmax,Ymin,Ymax,i,FailCount=0;
    COLOR_RGB Rgb;
    COLOR_HSL Hsl;

    for(i=oldx;i>IMG_X;i--)
    {
        ReadColor(i,oldy,&Rgb);
        RGBtoHSL(&Rgb,&Hsl);
        if(!ColorMatch(&Hsl,Condition))
            FailCount++;
        if(FailCount>(((Condition->WIDTH_MIN+Condition->WIDTH_MAX)>>2)>>ALLOW_FAIL_PER))
            break;
    }
    Xmin=i;
    FailCount=0;

    for(i=oldx;i<IMG_X+IMG_W;i++)
    {
        ReadColor(i,oldy,&Rgb);
        RGBtoHSL(&Rgb,&Hsl);
        if(!ColorMatch(&Hsl,Condition))
            FailCount++;
        if(FailCount>(((Condition->WIDTH_MIN+Condition->WIDTH_MAX)>>2)>>ALLOW_FAIL_PER))
            break;
    }

    Xmax=i;
    FailCount=0;

    for(i=oldy;i>IMG_Y;i--)
    {
        ReadColor(oldx,i,&Rgb);
        RGBtoHSL(&Rgb,&Hsl);
        if(!ColorMatch(&Hsl,Condition))
            FailCount++;
        if(FailCount>(((Condition->HIGHT_MIN+Condition->HIGHT_MAX)>>2)>>ALLOW_FAIL_PER))
            break;
    }

    Ymin=i;
    FailCount=0;

    for(i=oldy;i<IMG_Y+IMG_H;i++)
    {
        ReadColor(oldx,i,&Rgb);
        RGBtoHSL(&Rgb,&Hsl);
        if(!ColorMatch(&Hsl,Condition))
            FailCount++;
        if(FailCount>(((Condition->HIGHT_MIN+Condition->HIGHT_MAX)>>2)>>ALLOW_FAIL_PER))
            break;
    }

    Ymax=i;
    FailCount=0;

    Resu->x = (Xmin+Xmax)/2;
    Resu->y = (Ymin+Ymax)/2;
    Resu->w = Xmax-Xmin;
    Resu->h = Ymax-Ymin;

    if(((Xmax-Xmin)>(Condition->WIDTH_MIN)) && ((Ymax-Ymin)>(Condition->HIGHT_MIN)) &&
       ((Xmax-Xmin)<(Condition->WIDTH_MAX)) && ((Ymax-Ymin)<(Condition->HIGHT_MAX)) )
    {

        return 1;

    }
    else{
        ips200_show_int(16*6,16*10, Xmax-Xmin,3);
        printf("\nSB");
        return 0;

    }
}

//return 1 for success,0 for failure
int Trace(const TARGET_CONDI *Condition,RESULT *Resu)
{
    unsigned int i;
    static unsigned int x0,y0,flag=0;
    static SEARCH_AREA Area={IMG_X,IMG_X+IMG_W,IMG_Y,IMG_Y+IMG_H};
    RESULT Result;

    if(flag==0)
    {
        if(SearchCentre(&x0,&y0,Condition,&Area))
            flag=1;
        else
        {
            Area.X_Start= IMG_X ;
            Area.X_End  = IMG_X+IMG_W  ;
            Area.Y_Start= IMG_Y     ;
            Area.Y_End  = IMG_Y+IMG_H;

            if(SearchCentre(&x0,&y0,Condition,&Area))
            {
                flag=0;
                return 0;
            }
        }
    }
    Result.x = x0;
    Result.y = y0;

    for(i=0;i<ITERATE_NUM;i++)
        Corrode(Result.x,Result.y,Condition,&Result);

    if(Corrode(Result.x,Result.y,Condition,&Result))
    {
        x0=Result.x;
        y0=Result.y;
        Resu->x=Result.x;
        Resu->y=Result.y;
        Resu->w=Result.w;
        Resu->h=Result.h;
        flag=1;

        Area.X_Start= Result.x - ((Result.w)>>1);
        Area.X_End  = Result.x + ((Result.w)>>1);
        Area.Y_Start= Result.y - ((Result.h)>>1);
        Area.Y_End  = Result.y + ((Result.h)>>1);

        return 1;
    }
    else
    {

        flag=0;
        return 0;
    }

}

void RGB565_to_RGB888(unsigned short rgb565, COLOR_RGB *rgb)
{
#if SCC8660_SWAP_BYTES
    rgb565 = ((rgb565 & 0xFF) << 8) | ((rgb565 >> 8) & 0xFF);
#endif
    rgb->red   = (unsigned char)(((rgb565 >> 11) & 0x1F) * 255 / 31);
    rgb->green = (unsigned char)(((rgb565 >> 5)  & 0x3F) * 255 / 63);
    rgb->blue  = (unsigned char)(( rgb565        & 0x1F) * 255 / 31);
}

// ============================================================
//  2. RGB888 → HSL
// ============================================================
#define min3v(v1,v2,v3)  ((v1)>(v2)? ((v2)>(v3)?(v3):(v2)) : ((v1)>(v3)?(v3):(v1)))
#define max3v(v1,v2,v3)  ((v1)<(v2)? ((v2)<(v3)?(v3):(v2)) : ((v1)<(v3)?(v3):(v1)))

void RGB888_to_HSL(const COLOR_RGB *rgb, COLOR_HSL *hsl)
{
    int r = rgb->red;
    int g = rgb->green;
    int b = rgb->blue;
    int maxVal = max3v(r, g, b);
    int minVal = min3v(r, g, b);
    int difVal = maxVal - minVal;
    int h, s, l;

    l = (maxVal + minVal) * 240 / 255 / 2;

    if (maxVal == minVal)
    {
        h = 0;
        s = 0;
    }
    else
    {
        if (maxVal == r)
        {
            if (g >= b) h = 40 * (g - b) / difVal;
            else       h = 40 * (g - b) / difVal + 240;
        }
        else if (maxVal == g)
            h = 40 * (b - r) / difVal + 80;
        else
            h = 40 * (r - g) / difVal + 160;

        if (l == 0)
            s = 0;
        else if (l <= 120)
            s = difVal * 240 / (maxVal + minVal);
        else
            s = difVal * 240 / (480 - (maxVal + minVal));
    }

    hsl->hue        = (unsigned char)((h > 240) ? 240 : ((h < 0) ? 0 : h));
    hsl->saturation = (unsigned char)((s > 240) ? 240 : ((s < 0) ? 0 : s));
    hsl->luminance  = (unsigned char)((l > 240) ? 240 : ((l < 0) ? 0 : l));
}

// ============================================================
//  6. BFS 洪水填充 (HSL版)
//     对匹配像素累加RGB统计, 用于后续计算平均色
// ============================================================
static int FloodFillHSL(unsigned int seed_x, unsigned int seed_y,
                         unsigned char h_min, unsigned char h_max,
                         unsigned char s_min, unsigned char s_max,
                         unsigned char l_min, unsigned char l_max,
                         unsigned int max_pixels,
                         RESULT *res,
                         unsigned long *r_sum, unsigned long *g_sum,
                         unsigned long *b_sum, unsigned int *pixel_cnt)
{
    unsigned int head = 0, tail = 0;
    unsigned long rs = 0, gs = 0, bs = 0;
    unsigned int  cnt = 0;
    unsigned int  min_x = seed_x, max_x = seed_x;
    unsigned int  min_y = seed_y, max_y = seed_y;
    COLOR_RGB rgb;
    COLOR_HSL hsl;

    visit_clear();

    fill_qx[0] = (unsigned short)seed_x;
    fill_qy[0] = (unsigned short)seed_y;
    tail = 1;
    visit_set(seed_x, seed_y);

    while (head < tail && cnt < max_pixels)
    {
        unsigned int x = fill_qx[head];
        unsigned int y = fill_qy[head];
        head++;

        /* 读取像素并累加 */
        ReadColor(x, y, &rgb);
        rs += rgb.red;
        gs += rgb.green;
        bs += rgb.blue;
        cnt++;

        if (x < min_x) min_x = x;
        if (x > max_x) max_x = x;
        if (y < min_y) min_y = y;
        if (y > max_y) max_y = y;

        /* 四邻域扩展 */
        /* 左 */
        if (x > IMG_X && !visit_test(x - 1, y))
        {
            visit_set(x - 1, y);
            ReadColor(x - 1, y, &rgb);
            RGB888_to_HSL(&rgb, &hsl);
            if (ColorMatchHSL(&hsl, h_min, h_max, s_min, s_max, l_min, l_max)
                && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x - 1);
                fill_qy[tail] = (unsigned short)y;
                tail++;
            }
        }
        /* 右 */
        if (x + 1 < IMG_X + IMG_W && !visit_test(x + 1, y))
        {
            visit_set(x + 1, y);
            ReadColor(x + 1, y, &rgb);
            RGB888_to_HSL(&rgb, &hsl);
            if (ColorMatchHSL(&hsl, h_min, h_max, s_min, s_max, l_min, l_max)
                && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x + 1);
                fill_qy[tail] = (unsigned short)y;
                tail++;
            }
        }
        /* 上 */
        if (y > IMG_Y && !visit_test(x, y - 1))
        {
            visit_set(x, y - 1);
            ReadColor(x, y - 1, &rgb);
            RGB888_to_HSL(&rgb, &hsl);
            if (ColorMatchHSL(&hsl, h_min, h_max, s_min, s_max, l_min, l_max)
                && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x);
                fill_qy[tail] = (unsigned short)(y - 1);
                tail++;
            }
        }
        /* 下 */
        if (y + 1 < IMG_Y + IMG_H && !visit_test(x, y + 1))
        {
            visit_set(x, y + 1);
            ReadColor(x, y + 1, &rgb);
            RGB888_to_HSL(&rgb, &hsl);
            if (ColorMatchHSL(&hsl, h_min, h_max, s_min, s_max, l_min, l_max)
                && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x);
                fill_qy[tail] = (unsigned short)(y + 1);
                tail++;
            }
        }
    }

    if (cnt == 0) return 0;

    res->x = (min_x + max_x) / 2;
    res->y = (min_y + max_y) / 2;
    res->w = max_x - min_x + 1;
    res->h = max_y - min_y + 1;

    if (r_sum) *r_sum = rs;
    if (g_sum) *g_sum = gs;
    if (b_sum) *b_sum = bs;
    if (pixel_cnt) *pixel_cnt = cnt;

    return 1;
}

//  7. 网格搜索种子 (HSL版)
// ============================================================
static int GridSearchSeed(unsigned int *sx, unsigned int *sy,
                           unsigned char h_min, unsigned char h_max,
                           unsigned char s_min, unsigned char s_max,
                           unsigned char l_min, unsigned char l_max,
                           unsigned int step,
                           const SEARCH_AREA *area)
{
    unsigned int i, j, k, fail;
    unsigned int half = step / 2;
    COLOR_RGB rgb;
    COLOR_HSL hsl;

    if (step < 4) step = 4;

    for (i = area->Y_Start; i < area->Y_End; i += step)
    {
        for (j = area->X_Start; j < area->X_End; j += step)
        {
            fail = 0;

            /* 十字线检测 */
            for (k = 0; k < step * 2; k++)
            {
                unsigned int px, py;
                if (k < step)
                {
                    px = j + k;
                    py = i + half;
                }
                else
                {
                    px = j + half;
                    py = i + (k - step);
                }

                if (px >= IMG_X + IMG_W || py >= IMG_Y + IMG_H)
                {
                    fail++;
                    continue;
                }

                ReadColor(px, py, &rgb);
                RGB888_to_HSL(&rgb, &hsl);
                if (!ColorMatchHSL(&hsl, h_min, h_max, s_min, s_max, l_min, l_max))
                    fail++;

                if (fail > (step * 2) / 3)
                    break;
            }

            if (k == step * 2)
            {
                *sx = j + half;
                *sy = i + half;
                return 1;
            }
        }
    }
    return 0;
}

// ============================================================
//  7a. 参照色简化RGB匹配 (无HSL转换, 超快)
//      自动检测主通道, 适用于任意颜色参照物
// ============================================================
static int ColorMatchRefRGB(const COLOR_RGB *rgb)
{
    /* RGB 区间检查 */
    if (rgb->red   < ref_rgb_range.R_MIN || rgb->red   > ref_rgb_range.R_MAX) return 0;
    if (rgb->green < ref_rgb_range.G_MIN || rgb->green > ref_rgb_range.G_MAX) return 0;
    if (rgb->blue  < ref_rgb_range.B_MIN || rgb->blue  > ref_rgb_range.B_MAX) return 0;

    /* 主通道优势检查 (防颜色混叠) */
    if (ref_dominant_channel == 0)       /* R主导 (如红色衣物) */
    {
        if (rgb->red < rgb->green + 20) return 0;
        if (rgb->red < rgb->blue  + 20) return 0;
    }
    else if (ref_dominant_channel == 1)  /* G主导 (如绿色背景) */
    {
        if (rgb->green < rgb->red   + 20) return 0;
        if (rgb->green < rgb->blue  + 20) return 0;
    }
    else                                 /* B主导 (如蓝色背景) */
    {
        if (rgb->blue < rgb->red    + 20) return 0;
        if (rgb->blue < rgb->green + 20) return 0;
    }
    return 1;
}

// ============================================================
//  7b. 参照色 BFS 洪水填充 (仅粗填充, 简化RGB匹配)
//     无 HSL 转换 — 每像素约 30 周期 vs 目标色约 300 周期
//     res->x/y 存 min 坐标 (不是质心), 方便直接做搜索区域
// ============================================================
static int FloodFillRefRGB(unsigned int seed_x, unsigned int seed_y,
                            unsigned int max_pixels, RESULT *res)
{
    unsigned int head = 0, tail = 0;
    unsigned int cnt  = 0;
    unsigned int min_x = seed_x, max_x = seed_x;
    unsigned int min_y = seed_y, max_y = seed_y;
    COLOR_RGB rgb;

    visit_clear();

    fill_qx[0] = (unsigned short)seed_x;
    fill_qy[0] = (unsigned short)seed_y;
    tail = 1;
    visit_set(seed_x, seed_y);

    while (head < tail && cnt < max_pixels)
    {
        unsigned int x = fill_qx[head];
        unsigned int y = fill_qy[head];
        head++;
        cnt++;

        if (x < min_x) min_x = x;
        if (x > max_x) max_x = x;
        if (y < min_y) min_y = y;
        if (y > max_y) max_y = y;

        /* 四邻域扩展 — 仅 RGB 匹配, 不做 HSL 转换 */
        /* 左 */
        if (x > IMG_X && !visit_test(x - 1, y))
        {
            visit_set(x - 1, y);
            ReadColor(x - 1, y, &rgb);
            if (ColorMatchRefRGB(&rgb) && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x - 1);
                fill_qy[tail] = (unsigned short)y;
                tail++;
            }
        }
        /* 右 */
        if (x + 1 < IMG_X + IMG_W && !visit_test(x + 1, y))
        {
            visit_set(x + 1, y);
            ReadColor(x + 1, y, &rgb);
            if (ColorMatchRefRGB(&rgb) && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x + 1);
                fill_qy[tail] = (unsigned short)y;
                tail++;
            }
        }
        /* 上 */
        if (y > IMG_Y && !visit_test(x, y - 1))
        {
            visit_set(x, y - 1);
            ReadColor(x, y - 1, &rgb);
            if (ColorMatchRefRGB(&rgb) && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x);
                fill_qy[tail] = (unsigned short)(y - 1);
                tail++;
            }
        }
        /* 下 */
        if (y + 1 < IMG_Y + IMG_H && !visit_test(x, y + 1))
        {
            visit_set(x, y + 1);
            ReadColor(x, y + 1, &rgb);
            if (ColorMatchRefRGB(&rgb) && tail < FILL_Q_SIZE)
            {
                fill_qx[tail] = (unsigned short)(x);
                fill_qy[tail] = (unsigned short)(y + 1);
                tail++;
            }
        }
    }

    if (cnt == 0) return 0;

    /* 返回 min/max 包围盒 (注意: 不是质心!) */
    res->x = min_x;
    res->y = min_y;
    res->w = max_x - min_x + 1;
    res->h = max_y - min_y + 1;
    return 1;
}

// ============================================================
//  7c. 参照色网格搜索种子
//      十字线检测 + 简化RGB匹配
// ============================================================
static int GridSearchRefSeed(unsigned int *sx, unsigned int *sy,
                              unsigned int step, const SEARCH_AREA *area)
{
    unsigned int i, j, k, fail;
    unsigned int half = step / 2;
    COLOR_RGB rgb;

    if (step < 4) step = 4;

    for (i = area->Y_Start; i < area->Y_End; i += step)
    {
        for (j = area->X_Start; j < area->X_End; j += step)
        {
            fail = 0;
            for (k = 0; k < step * 2; k++)
            {
                unsigned int px, py;
                if (k < step) { px = j + k;   py = i + half; }
                else          { px = j + half; py = i + (k - step); }

                if (px >= IMG_X + IMG_W || py >= IMG_Y + IMG_H)
                { fail++; continue; }

                ReadColor(px, py, &rgb);
                if (!ColorMatchRefRGB(&rgb)) fail++;
                if (fail > (step * 2) / 3) break;
            }
            if (k == step * 2)
            {
                *sx = j + half;
                *sy = i + half;
                return 1;
            }
        }
    }
    return 0;
}

// ============================================================
//  7d. FindRefColorBlock: 参照色快速搜索 (仅粗填充)
//      返回包围盒 (min/max 坐标格式)
//      用于约束目标色搜索范围
// ============================================================
static int FindRefColorBlock(RESULT *ref_res)
{
    static TRACE_STATE  ref_state = TS_FIND_IN_FULL_FRAME;
    static RESULT       last_ref;
    static SEARCH_AREA  last_area;

    const SEARCH_AREA full_area = {CAM_X_START, CAM_X_END, IMG_Y, IMG_Y + IMG_H};
    unsigned int seed_x = 80, seed_y = 60;

    if (!calib_ref_valid) return 0;

    switch (ref_state)
    {
    case TS_TRACING:
        /* 上一帧参照色中心 */
        seed_x = last_ref.x + last_ref.w / 2;
        seed_y = last_ref.y + last_ref.h / 2;
        {
            COLOR_RGB srgb;
            ReadColor(seed_x, seed_y, &srgb);
            if (ColorMatchRefRGB(&srgb) &&
                FloodFillRefRGB(seed_x, seed_y, REF_FILL_MAX_PIXELS, ref_res))
            { break; }
        }
        ref_state = TS_FIND_IN_LAST_FRAME;
        /* FALLTHROUGH */

    case TS_FIND_IN_LAST_FRAME:
        if (GridSearchRefSeed(&seed_x, &seed_y, 12, &last_area) &&
            FloodFillRefRGB(seed_x, seed_y, REF_FILL_MAX_PIXELS, ref_res))
        {
            ref_state = TS_TRACING;
            break;
        }
        ref_state = TS_FIND_IN_FULL_FRAME;
        /* FALLTHROUGH */

    case TS_FIND_IN_FULL_FRAME:
        if (GridSearchRefSeed(&seed_x, &seed_y, 12, &full_area) &&
            FloodFillRefRGB(seed_x, seed_y, REF_FILL_MAX_PIXELS, ref_res))
        {
            ref_state = TS_TRACING;
            break;
        }
        ref_res->x = ref_res->y = ref_res->w = ref_res->h = 0;
        return 0;
    }

    /* 更新追踪 */
    last_ref = *ref_res;
    last_area.X_Start = (ref_res->x > ref_res->w) ? (ref_res->x - ref_res->w) : CAM_X_START;
    last_area.X_End   = (ref_res->x + ref_res->w * 2 < CAM_X_END) ? (ref_res->x + ref_res->w * 2) : CAM_X_END;
    last_area.Y_Start = (ref_res->y > ref_res->h) ? (ref_res->y - ref_res->h) : IMG_Y;
    last_area.Y_End   = (ref_res->y + ref_res->h * 2 < IMG_Y + IMG_H) ? (ref_res->y + ref_res->h * 2) : IMG_Y + IMG_H;
    ref_state = TS_TRACING;

    return 1;
}

// ============================================================
//  7e. SpatialVerify: 空间约束验证
//      目标质心是否在参照色包围盒内 (包含约束)
//      允许 5px 容差 (边缘像素不稳定的 BFS 边界)
// ============================================================
#define SPATIAL_TOLERANCE  5
static int SpatialVerify(const RESULT *ref, const RESULT *target)
{
    int t_cx = (int)target->x;
    int t_cy = (int)target->y;
    int r_left   = (int)ref->x - SPATIAL_TOLERANCE;
    int r_right  = (int)(ref->x + ref->w) + SPATIAL_TOLERANCE;
    int r_top    = (int)ref->y - SPATIAL_TOLERANCE;
    int r_bottom = (int)(ref->y + ref->h) + SPATIAL_TOLERANCE;

    return (t_cx >= r_left && t_cx <= r_right &&
            t_cy >= r_top  && t_cy <= r_bottom);
}

// ============================================================
//  8. CalibrateFromCenter
//     采样中心7×7 → 平均RGB → 宽松HSL阈值 → 保存Flash
//     (双色模式: 第一次校准=参照色, 第二次校准=目标色)
// ============================================================
void CalibrateFromCenter(TARGET_CONDI *cond)
{
    int cx = IMG_W / 2, cy = IMG_H / 2;
    int r = 3, i, j, cnt = 0;
    unsigned long r_sum = 0, g_sum = 0, b_sum = 0;
    COLOR_RGB rgb;
    COLOR_HSL hsl;
    int avg_r, avg_g, avg_b;

    for (i = cy - r; i <= cy + r; i++)
    {
        for (j = cx - r; j <= cx + r; j++)
        {
            ReadColor((unsigned int)j, (unsigned int)i, &rgb);
            r_sum += rgb.red;
            g_sum += rgb.green;
            b_sum += rgb.blue;
            cnt++;
        }
    }

    avg_r = (int)(r_sum / cnt);
    avg_g = (int)(g_sum / cnt);
    avg_b = (int)(b_sum / cnt);

    /* ======== 双色校准状态机 ======== */
    if (calib_state == CALIB_STATE_NONE || calib_state == CALIB_STATE_DONE)
    {
        /* ---- 阶段1: 校准参照色 (背景/红色衣物) ---- */

        calib_ref_rgb.red   = (unsigned char)avg_r;
        calib_ref_rgb.green = (unsigned char)avg_g;
        calib_ref_rgb.blue  = (unsigned char)avg_b;

        /* 设定参照色简化RGB阈值 (宽松) */
        ref_rgb_range.R_MIN = (unsigned char)((avg_r > REF_RGB_TOL) ? (avg_r - REF_RGB_TOL) : 0);
        ref_rgb_range.R_MAX = (unsigned char)((avg_r + REF_RGB_TOL < 255) ? (avg_r + REF_RGB_TOL) : 255);
        ref_rgb_range.G_MIN = (unsigned char)((avg_g > REF_RGB_TOL) ? (avg_g - REF_RGB_TOL) : 0);
        ref_rgb_range.G_MAX = (unsigned char)((avg_g + REF_RGB_TOL < 255) ? (avg_g + REF_RGB_TOL) : 255);
        ref_rgb_range.B_MIN = (unsigned char)((avg_b > REF_RGB_TOL) ? (avg_b - REF_RGB_TOL) : 0);
        ref_rgb_range.B_MAX = (unsigned char)((avg_b + REF_RGB_TOL < 255) ? (avg_b + REF_RGB_TOL) : 255);

        /* 自动检测主通道 */
        if (avg_r >= avg_g && avg_r >= avg_b)
            ref_dominant_channel = 0;   /* R 主导 */
        else if (avg_g >= avg_r && avg_g >= avg_b)
            ref_dominant_channel = 1;   /* G 主导 */
        else
            ref_dominant_channel = 2;   /* B 主导 */

        calib_ref_valid = 1;
        calib_state = CALIB_STATE_REF_DONE;

        /* 保存参照色到 Flash Page10 */
        CalibSaveRefToFlash();

        printf("\n[校准-参照色] RGB: R=%d G=%d B=%d  主通道:%d\n",
               avg_r, avg_g, avg_b, ref_dominant_channel);
        printf("  阈值: R[%d,%d] G[%d,%d] B[%d,%d]\n",
               ref_rgb_range.R_MIN, ref_rgb_range.R_MAX,
               ref_rgb_range.G_MIN, ref_rgb_range.G_MAX,
               ref_rgb_range.B_MIN, ref_rgb_range.B_MAX);
    }
    else if (calib_state == CALIB_STATE_REF_DONE)
    {
        /* ---- 阶段2: 校准目标色 (前景/黄色贴纸) ---- */

        calib_target.red   = (unsigned char)avg_r;
        calib_target.green = (unsigned char)avg_g;
        calib_target.blue  = (unsigned char)avg_b;

        /* 转为HSL → 设定宽松阈值 */
        RGB888_to_HSL(&calib_target, &hsl);

        calib_H_min = (unsigned char)((hsl.hue > CALIB_H_TOL) ? (hsl.hue - CALIB_H_TOL) : 0);
        calib_H_max = (unsigned char)((hsl.hue + CALIB_H_TOL < 240) ? (hsl.hue + CALIB_H_TOL) : 240);
        calib_S_min = (unsigned char)((hsl.saturation > CALIB_S_TOL) ? (hsl.saturation - CALIB_S_TOL) : 0);
        calib_S_max = (unsigned char)((hsl.saturation + CALIB_S_TOL < 240) ? (hsl.saturation + CALIB_S_TOL) : 240);
        calib_L_min = (unsigned char)((hsl.luminance > CALIB_L_TOL) ? (hsl.luminance - CALIB_L_TOL) : 0);
        calib_L_max = (unsigned char)((hsl.luminance + CALIB_L_TOL < 240) ? (hsl.luminance + CALIB_L_TOL) : 240);
        calib_valid = 1;
        calib_state = CALIB_STATE_DONE;

        /* 同步到 cond (用于屏幕显示) */
        cond->H_MIN = calib_H_min;  cond->H_MAX = calib_H_max;
        cond->S_MIN = calib_S_min;  cond->S_MAX = calib_S_max;
        cond->L_MIN = calib_L_min;  cond->L_MAX = calib_L_max;
        cond->WIDTH_MIN  = 3;   cond->HIGHT_MIN  = 3;
        cond->WIDTH_MAX  = 100; cond->HIGHT_MAX  = 100;

        /* 保存目标色到 Flash Page11 */
        CalibSaveToFlash(cond);

        printf("\n[校准-目标色] RGB: R=%d G=%d B=%d\n",
               calib_target.red, calib_target.green, calib_target.blue);
        printf("  宽松: H[%d,%d] S[%d,%d] L[%d,%d]\n",
               calib_H_min, calib_H_max, calib_S_min, calib_S_max, calib_L_min, calib_L_max);
    }
}

// ============================================================
//  9a. 保存到 Flash
// ============================================================
void CalibSaveToFlash(const TARGET_CONDI *cond)
{
    uint32 buf[8];

    buf[0] = CALIB_MAGIC;
    buf[1] = ((uint32)calib_target.red)        |
             ((uint32)calib_target.green << 8)  |
             ((uint32)calib_target.blue  << 16) |
             ((uint32)calib_H_min << 24);
    buf[2] = ((uint32)calib_H_max)       |
             ((uint32)calib_S_min << 8)   |
             ((uint32)calib_S_max << 16)  |
             ((uint32)calib_L_min << 24);
    buf[3] = calib_L_max;
    buf[4] = cond->WIDTH_MIN;
    buf[5] = cond->HIGHT_MIN;
    buf[6] = cond->WIDTH_MAX;
    buf[7] = cond->HIGHT_MAX;

    flash_erase_page(0, CALIB_FLASH_PAGE);
    flash_write_page(0, CALIB_FLASH_PAGE, buf, 8);
}

// ============================================================
//  9b. 从 Flash 加载
// ============================================================
int CalibLoadFromFlash(TARGET_CONDI *cond)
{
    uint32 buf[8];
    flash_read_page(0, CALIB_FLASH_PAGE, buf, 8);

    if (buf[0] != CALIB_MAGIC) return 0;

    calib_target.red   = (unsigned char)( buf[1]        & 0xFF);
    calib_target.green = (unsigned char)((buf[1] >> 8)  & 0xFF);
    calib_target.blue  = (unsigned char)((buf[1] >> 16) & 0xFF);
    calib_H_min = (unsigned char)((buf[1] >> 24) & 0xFF);

    calib_H_max = (unsigned char)( buf[2]        & 0xFF);
    calib_S_min = (unsigned char)((buf[2] >> 8)  & 0xFF);
    calib_S_max = (unsigned char)((buf[2] >> 16) & 0xFF);
    calib_L_min = (unsigned char)((buf[2] >> 24) & 0xFF);

    calib_L_max     = (unsigned char)(buf[3] & 0xFF);
    cond->WIDTH_MIN  = buf[4];
    cond->HIGHT_MIN  = buf[5];
    cond->WIDTH_MAX  = buf[6];
    cond->HIGHT_MAX  = buf[7];

    /* 同步到 cond 用于屏幕显示 */
    cond->H_MIN = calib_H_min;  cond->H_MAX = calib_H_max;
    cond->S_MIN = calib_S_min;  cond->S_MAX = calib_S_max;
    cond->L_MIN = calib_L_min;  cond->L_MAX = calib_L_max;

    calib_valid = 1;
    return 1;
}

// ============================================================
//  9c. 保存参照色到 Flash Page10
//      复用与目标色相同的 8-word 格式, 独立存储
// ============================================================
static void CalibSaveRefToFlash(void)
{
    uint32 buf[8];

    buf[0] = CALIB_MAGIC_REF;
    buf[1] = ((uint32)calib_ref_rgb.red)           |
             ((uint32)calib_ref_rgb.green    << 8)  |
             ((uint32)calib_ref_rgb.blue     << 16) |
             ((uint32)ref_rgb_range.R_MIN    << 24);
    buf[2] = ((uint32)ref_rgb_range.R_MAX)          |
             ((uint32)ref_rgb_range.G_MIN    << 8)   |
             ((uint32)ref_rgb_range.G_MAX    << 16)  |
             ((uint32)ref_rgb_range.B_MIN    << 24);
    buf[3] = ((uint32)ref_rgb_range.B_MAX)          |
             ((uint32)ref_dominant_channel   << 8);
    buf[4] = 0;  /* 预留 */
    buf[5] = 0;
    buf[6] = 0;
    buf[7] = 0;

    flash_erase_page(0, CALIB_FLASH_PAGE_REF);
    flash_write_page(0, CALIB_FLASH_PAGE_REF, buf, 8);
}

// ============================================================
//  9d. 从 Flash Page10 加载参照色
// ============================================================
static int CalibLoadRefFromFlash(void)
{
    uint32 buf[8];
    flash_read_page(0, CALIB_FLASH_PAGE_REF, buf, 8);

    if (buf[0] != CALIB_MAGIC_REF) return 0;

    calib_ref_rgb.red   = (unsigned char)( buf[1]        & 0xFF);
    calib_ref_rgb.green = (unsigned char)((buf[1] >> 8)  & 0xFF);
    calib_ref_rgb.blue  = (unsigned char)((buf[1] >> 16) & 0xFF);
    ref_rgb_range.R_MIN = (unsigned char)((buf[1] >> 24) & 0xFF);

    ref_rgb_range.R_MAX = (unsigned char)( buf[2]        & 0xFF);
    ref_rgb_range.G_MIN = (unsigned char)((buf[2] >> 8)  & 0xFF);
    ref_rgb_range.G_MAX = (unsigned char)((buf[2] >> 16) & 0xFF);
    ref_rgb_range.B_MIN = (unsigned char)((buf[2] >> 24) & 0xFF);

    ref_rgb_range.B_MAX  = (unsigned char)( buf[3]        & 0xFF);
    ref_dominant_channel = (int)((buf[3] >> 8) & 0xFF);

    calib_ref_valid = 1;
    return 1;
}

// ============================================================
//  10. FindColorBlock: 范式A双色追踪
//
//   参照色快速搜索(简化RGB) → 约束目标色搜索范围 → 空间验证
//   状态机: FULL_FRAME → LAST_FRAME → TRACING
//   第一步: FloodFillHSL(宽松HSL) → 圈范围 + 算均值
//   第二步: FloodFillHSL(收紧HSL) → 精确边界
//   前置: 参照色搜索 → 约束 SEARCH_AREA
// ============================================================
int FindColorBlock(RESULT *res, COLOR_RGB *avg_rgb)
{
    static TRACE_STATE  state = TS_FIND_IN_FULL_FRAME;
    static RESULT       last_res;
    static SEARCH_AREA  last_area;

    SEARCH_AREA search_area;
    const SEARCH_AREA full_area = {CAM_X_START, CAM_X_END, IMG_Y, IMG_Y + IMG_H};
    unsigned int seed_x = 80, seed_y = 60;
    RESULT      pass1;
    COLOR_RGB   region_avg;
    unsigned long r_sum, g_sum, b_sum;
    unsigned int  p_cnt;
    COLOR_HSL    avg_hsl;
    unsigned char tH_min, tH_max, tS_min, tS_max, tL_min, tL_max;
    int          seed_ok = 0;
    int          ref_found = 0;

    if (!calib_valid) {
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* ======== 阶段0: 参照色快速搜索 (范式A) ======== */
    if (calib_ref_valid)
    {
        ref_found = FindRefColorBlock(&g_last_ref_res);
        if (!ref_found)
        {
            /* 参照色丢失 → 严格模式: 不降级, 直接返回未找到 */
            state = TS_FIND_IN_FULL_FRAME;
            res->x = res->y = res->w = res->h = 0;
            return 0;
        }
        /* 用参照色包围盒约束目标色搜索范围 (+ margin) */
        search_area.X_Start = (g_last_ref_res.x > 10) ? (g_last_ref_res.x - 10) : CAM_X_START;
        search_area.Y_Start = (g_last_ref_res.y > 10) ? (g_last_ref_res.y - 10) : IMG_Y;
        search_area.X_End   = (g_last_ref_res.x + g_last_ref_res.w + 10 < CAM_X_END)
                              ? (g_last_ref_res.x + g_last_ref_res.w + 10) : CAM_X_END;
        search_area.Y_End   = (g_last_ref_res.y + g_last_ref_res.h + 10 < IMG_Y + IMG_H)
                              ? (g_last_ref_res.y + g_last_ref_res.h + 10) : IMG_Y + IMG_H;
    }
    else
    {
        /* 无参照色校准 → 降级全图搜索 (单色兼容模式) */
        search_area = full_area;
    }

    /* ======== 阶段1: 寻找种子 ======== */
    switch (state)
    {
    case TS_TRACING:
        seed_x = last_res.x;
        seed_y = last_res.y;
        {
            COLOR_RGB srgb;
            COLOR_HSL shsl;
            ReadColor(seed_x, seed_y, &srgb);
            RGB888_to_HSL(&srgb, &shsl);
            if (ColorMatchHSL(&shsl, calib_H_min, calib_H_max,
                              calib_S_min, calib_S_max,
                              calib_L_min, calib_L_max))
            {
                seed_ok = 1;
                break;
            }
        }
        state = TS_FIND_IN_LAST_FRAME;
        /* FALLTHROUGH */

    case TS_FIND_IN_LAST_FRAME:
        if (GridSearchSeed(&seed_x, &seed_y, calib_H_min, calib_H_max,
                           calib_S_min, calib_S_max, calib_L_min, calib_L_max,
                           12, &last_area))
        {
            seed_ok = 1;
            state = TS_TRACING;
            break;
        }
        state = TS_FIND_IN_FULL_FRAME;
        /* FALLTHROUGH */

    case TS_FIND_IN_FULL_FRAME:
        if (GridSearchSeed(&seed_x, &seed_y, calib_H_min, calib_H_max,
                           calib_S_min, calib_S_max, calib_L_min, calib_L_max,
                           12, &search_area))
        {
            seed_ok = 1;
            state = TS_TRACING;
            break;
        }
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    if (!seed_ok) {
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* ======== 阶段2: 第一步 — 粗填充 (宽松阈值) ======== */
    if (!FloodFillHSL(seed_x, seed_y,
                      calib_H_min, calib_H_max,
                      calib_S_min, calib_S_max,
                      calib_L_min, calib_L_max,
                      MAX_FILL_PIXELS, &pass1,
                      &r_sum, &g_sum, &b_sum, &p_cnt))
    {
        state = TS_FIND_IN_LAST_FRAME;
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* 计算区域平均RGB → HSL → 收紧阈值 */
    region_avg.red   = (unsigned char)(r_sum / p_cnt);
    region_avg.green = (unsigned char)(g_sum / p_cnt);
    region_avg.blue  = (unsigned char)(b_sum / p_cnt);
    RGB888_to_HSL(&region_avg, &avg_hsl);

    tH_min = (unsigned char)((avg_hsl.hue > TIGHT_H_TOL) ? (avg_hsl.hue - TIGHT_H_TOL) : 0);
    tH_max = (unsigned char)((avg_hsl.hue + TIGHT_H_TOL < 240) ? (avg_hsl.hue + TIGHT_H_TOL) : 240);
    tS_min = (unsigned char)((avg_hsl.saturation > TIGHT_S_TOL) ? (avg_hsl.saturation - TIGHT_S_TOL) : 0);
    tS_max = (unsigned char)((avg_hsl.saturation + TIGHT_S_TOL < 240) ? (avg_hsl.saturation + TIGHT_S_TOL) : 240);
    tL_min = (unsigned char)((avg_hsl.luminance > TIGHT_L_TOL) ? (avg_hsl.luminance - TIGHT_L_TOL) : 0);
    tL_max = (unsigned char)((avg_hsl.luminance + TIGHT_L_TOL < 240) ? (avg_hsl.luminance + TIGHT_L_TOL) : 240);

    /* ======== 阶段3: 第二步 — 精填充 (收紧阈值) ======== */
    if (!FloodFillHSL(pass1.x, pass1.y,
                      tH_min, tH_max, tS_min, tS_max, tL_min, tL_max,
                      MAX_FILL_PIXELS, res,
                      NULL, NULL, NULL, NULL))
    {
        state = TS_FIND_IN_LAST_FRAME;
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* ======== 阶段3.5: 空间验证 (范式A) ======== */
    if (ref_found && !SpatialVerify(&g_last_ref_res, res))
    {
        /* 目标找到但不在参照区域内 → 干扰, 拒绝 */
        state = TS_FIND_IN_LAST_FRAME;
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* ======== 验证尺寸 ======== */
    if (res->w < 2 || res->h < 2 || res->w > 160 || res->h > 120)
    {
        state = TS_FIND_IN_LAST_FRAME;
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* ======== 输出平均色 ======== */
    if (avg_rgb) *avg_rgb = region_avg;

    /* ======== 更新追踪 ======== */
    last_res = *res;
    {
        unsigned int lw = res->w;
        unsigned int lh = res->h;
        last_area.X_Start = (res->x > lw) ? (res->x - lw) : CAM_X_START;
        last_area.X_End   = (res->x + lw < CAM_X_END) ? (res->x + lw) : CAM_X_END;
        last_area.Y_Start = (res->y > lh) ? (res->y - lh) : IMG_Y;
        last_area.Y_End   = (res->y + lh < IMG_Y + IMG_H) ? (res->y + lh) : (IMG_Y + IMG_H);
    }
    state = TS_TRACING;

    return 1;
}

/==================== 8. CalibrateFromCenter：校准函数（双色状态机） ====================
/*
 * 从画面中心 7×7 区域采样平均 RGB，然后使用洪水填充扩大采样范围，
 * 以覆盖褶皱暗面/亮面，得到更稳定的平均色。
 * 根据 calib_state 状态机决定当前校准的是参照色还是目标色：
 *   - CALIB_STATE_NONE 或 DONE → 开始新一轮校准：先校准参照色（状态变为 REF_DONE）
 *   - CALIB_STATE_REF_DONE → 校准目标色，完成后轮次推进，若未满 5 组则回到 NONE。
 * 校准后的数据存入对应轮次的数组，并保存第 0 组到 Flash。
 * 参数 cond 用于回传目标色阈值（屏幕显示用）。
 */
void CalibrateFromCenter (TARGET_CONDI *cond)
{
    int cx = IMG_W / 2, cy = IMG_H / 2;
    int r = 3, i, j, cnt = 0;
    unsigned long r_sum = 0, g_sum = 0, b_sum = 0;
    COLOR_RGB rgb;
    COLOR_HSL hsl;
    int avg_r, avg_g, avg_b;

    /* ---- 1. 中心 7×7 采样 ---- */
    for (i = cy - r; i <= cy + r; i++)
    {
        for (j = cx - r; j <= cx + r; j++)
        {
            ReadColor((unsigned int)j, (unsigned int)i, &rgb);
            r_sum += rgb.red;
            g_sum += rgb.green;
            b_sum += rgb.blue;
            cnt++;
        }
    }
    avg_r = (int)(r_sum / cnt);
    avg_g = (int)(g_sum / cnt);
    avg_b = (int)(b_sum / cnt);

    /* ---- 2. 用洪水填充扩大采样（若成功且面积>49） ---- */
    {
        COLOR_RGB seed_rgb;
        COLOR_HSL seed_hsl;
        RESULT    fill_res;
        unsigned long fr_sum, fg_sum, fb_sum;
        unsigned int  fp_cnt;
        seed_rgb.red   = (unsigned char)avg_r;
        seed_rgb.green = (unsigned char)avg_g;
        seed_rgb.blue  = (unsigned char)avg_b;
        RGB888_to_HSL(&seed_rgb, &seed_hsl);

        /* 超宽容差：H 锁颜色，S/L 覆盖大面积明暗变化 */
        {
            unsigned char fH_min = (unsigned char)((seed_hsl.hue > CALIB_FLOOD_H_TOL)
                                    ? (seed_hsl.hue - CALIB_FLOOD_H_TOL) : 0);
            unsigned char fH_max = (unsigned char)((seed_hsl.hue + CALIB_FLOOD_H_TOL < 240)
                                    ? (seed_hsl.hue + CALIB_FLOOD_H_TOL) : 240);
            unsigned char fS_min = (unsigned char)((seed_hsl.saturation > CALIB_FLOOD_S_TOL)
                                    ? (seed_hsl.saturation - CALIB_FLOOD_S_TOL) : 0);
            unsigned char fS_max = (unsigned char)((seed_hsl.saturation + CALIB_FLOOD_S_TOL < 240)
                                    ? (seed_hsl.saturation + CALIB_FLOOD_S_TOL) : 240);
            unsigned char fL_min = (unsigned char)((seed_hsl.luminance > CALIB_FLOOD_L_TOL)
                                    ? (seed_hsl.luminance - CALIB_FLOOD_L_TOL) : 0);
            unsigned char fL_max = (unsigned char)((seed_hsl.luminance + CALIB_FLOOD_L_TOL < 240)
                                    ? (seed_hsl.luminance + CALIB_FLOOD_L_TOL) : 240);

            if (FloodFillHSL ((unsigned int) cx, (unsigned int) cy,
                             fH_min, fH_max, fS_min, fS_max, fL_min, fL_max,
                             MAX_FILL_PIXELS, &fill_res,
                             &fr_sum, &fg_sum, &fb_sum, &fp_cnt)
                && fp_cnt > 49)   // 必须比 7×7=49 大才有效
            {
                avg_r = (int)(fr_sum / fp_cnt);
                avg_g = (int)(fg_sum / fp_cnt);
                avg_b = (int)(fb_sum / fp_cnt);
            }
            // 若失败，回退到 7×7 均值
        }
    }

    /* ======== 双色校准状态机 ======== */
    if (calib_state == CALIB_STATE_NONE || calib_state == CALIB_STATE_DONE)
    {
        /* ---- 阶段 1: 校准参照色（背景）第 calib_round+1 轮 ---- */
        if (calib_round >= NUM_THRESHOLD_SETS)
            calib_round = 0;   // 若已完成 5 轮，重新从第 1 轮开始

        calib_ref_rgb.red   = (unsigned char)avg_r;
        calib_ref_rgb.green = (unsigned char)avg_g;
        calib_ref_rgb.blue  = (unsigned char)avg_b;

        /* 设定参照色简化 RGB 阈值（宽松） */
        ref_rgb_range.R_MIN = (unsigned char)((avg_r > REF_RGB_TOL) ? (avg_r - REF_RGB_TOL) : 0);
        ref_rgb_range.R_MAX = (unsigned char)((avg_r + REF_RGB_TOL < 255) ? (avg_r + REF_RGB_TOL) : 255);
        ref_rgb_range.G_MIN = (unsigned char)((avg_g > REF_RGB_TOL) ? (avg_g - REF_RGB_TOL) : 0);
        ref_rgb_range.G_MAX = (unsigned char)((avg_g + REF_RGB_TOL < 255) ? (avg_g + REF_RGB_TOL) : 255);
        ref_rgb_range.B_MIN = (unsigned char)((avg_b > REF_RGB_TOL) ? (avg_b - REF_RGB_TOL) : 0);
        ref_rgb_range.B_MAX = (unsigned char)((avg_b + REF_RGB_TOL < 255) ? (avg_b + REF_RGB_TOL) : 255);

        /* 自动检测主通道 */
        if (avg_r >= avg_g && avg_r >= avg_b)
            ref_dominant_channel = 0;   // R 主导
        else if (avg_g >= avg_r && avg_g >= avg_b)
            ref_dominant_channel = 1;   // G 主导
        else
            ref_dominant_channel = 2;   // B 主导

        calib_ref_valid = 1;
        calib_state = CALIB_STATE_REF_DONE;

        /* 存入对应轮次的数组 */
        saved_ref_rgb [calib_round]      = calib_ref_rgb;
        saved_ref_range [calib_round]    = ref_rgb_range;
        saved_ref_dominant [calib_round] = ref_dominant_channel;
        saved_ref_valid [calib_round]    = 1;

        CalibSaveRefToFlash();  // 仅第 0 组保存

        printf ("\n [校准 - 参照色 %d/%d] RGB: R=%d G=%d B=%d  主通道:%d\n",
               calib_round + 1, NUM_THRESHOLD_SETS,
               avg_r, avg_g, avg_b, ref_dominant_channel);
        printf (" 阈值: R [%d,%d] G [%d,%d] B [%d,%d]\n",
               ref_rgb_range.R_MIN, ref_rgb_range.R_MAX,
               ref_rgb_range.G_MIN, ref_rgb_range.G_MAX,
               ref_rgb_range.B_MIN, ref_rgb_range.B_MAX);
    }
    else if (calib_state == CALIB_STATE_REF_DONE)
    {
        /* ---- 阶段 2: 校准目标色（前景）第 calib_round+1 轮 ---- */
        calib_target.red   = (unsigned char)avg_r;
        calib_target.green = (unsigned char)avg_g;
        calib_target.blue  = (unsigned char)avg_b;

        /* 转为 HSL，设定宽松阈值（用于粗填充） */
        RGB888_to_HSL (&calib_target, &hsl);
        calib_H_min = (unsigned char)((hsl.hue > CALIB_H_TOL) ? (hsl.hue - CALIB_H_TOL) : 0);
        calib_H_max = (unsigned char)((hsl.hue + CALIB_H_TOL < 240) ? (hsl.hue + CALIB_H_TOL) : 240);
        calib_S_min = (unsigned char)((hsl.saturation > CALIB_S_TOL) ? (hsl.saturation - CALIB_S_TOL) : 0);
        calib_S_max = (unsigned char)((hsl.saturation + CALIB_S_TOL < 240) ? (hsl.saturation + CALIB_S_TOL) : 240);
        calib_L_min = (unsigned char)((hsl.luminance > CALIB_L_TOL) ? (hsl.luminance - CALIB_L_TOL) : 0);
        calib_L_max = (unsigned char)((hsl.luminance + CALIB_L_TOL < 240) ? (hsl.luminance + CALIB_L_TOL) : 240);
        calib_valid = 1;

        /* 存入对应轮次的数组 */
        saved_tgt_rgb [calib_round]   = calib_target;
        saved_tgt_H_min [calib_round] = calib_H_min;  saved_tgt_H_max [calib_round] = calib_H_max;
        saved_tgt_S_min [calib_round] = calib_S_min;  saved_tgt_S_max [calib_round] = calib_S_max;
        saved_tgt_L_min [calib_round] = calib_L_min;  saved_tgt_L_max [calib_round] = calib_L_max;
        saved_tgt_valid [calib_round] = 1;

        /* 同步到 cond（屏幕显示用） */
        cond->H_MIN = calib_H_min;  cond->H_MAX = calib_H_max;
        cond->S_MIN = calib_S_min;  cond->S_MAX = calib_S_max;
        cond->L_MIN = calib_L_min;  cond->L_MAX = calib_L_max;
        cond->WIDTH_MIN  = 3;   cond->HIGHT_MIN  = 3;
        cond->WIDTH_MAX  = 100; cond->HIGHT_MAX  = 100;

        CalibSaveToFlash (cond);  // 仅第 0 组保存

        printf ("\n [校准 - 目标色 %d/%d] RGB: R=%d G=%d B=%d\n",
               calib_round + 1, NUM_THRESHOLD_SETS,
               calib_target.red, calib_target.green, calib_target.blue);
        printf (" 宽松: H [%d,%d] S [%d,%d] L [%d,%d]\n",
               calib_H_min, calib_H_max, calib_S_min, calib_S_max, calib_L_min, calib_L_max);

        /* 轮次推进 */
        calib_round++;
        if (calib_round >= NUM_THRESHOLD_SETS)
            calib_state = CALIB_STATE_DONE;   // 5 轮全部完成
        else
            calib_state = CALIB_STATE_NONE;   // 还有下一轮，等待参照色校准
    }
}

//==================== 9a. 保存目标色到 Flash ====================
/*
 * 将当前目标色（第 0 组）的 HSL 阈值写入 Flash 页 11。
 * 仅当 calib_round == 0 时保存（仅持久化第 0 组）。
 */
void CalibSaveToFlash (const TARGET_CONDI *cond)
{
    uint32 buf [8];
    if (calib_round != 0) return;

    buf[0] = CALIB_MAGIC;
    buf[1] = ((uint32)calib_target.red)        |
             ((uint32)calib_target.green << 8)  |
             ((uint32)calib_target.blue  << 16) |
             ((uint32)calib_H_min << 24);
    buf[2] = ((uint32)calib_H_max)       |
             ((uint32)calib_S_min << 8)   |
             ((uint32)calib_S_max << 16)  |
             ((uint32)calib_L_min << 24);
    buf[3] = calib_L_max;
    buf[4] = cond->WIDTH_MIN;
    buf[5] = cond->HIGHT_MIN;
    buf[6] = cond->WIDTH_MAX;
    buf[7] = cond->HIGHT_MAX;

    flash_erase_page(0, CALIB_FLASH_PAGE_TGT);
    flash_write_page(0, CALIB_FLASH_PAGE_TGT, buf, 8);
}

//==================== 9b. 从 Flash 加载目标色 ====================
/*
 * 从 Flash 页 11 加载第 0 组目标色阈值，恢复工作变量。
 * 返回 1 成功，0 失败（无有效数据）。
 */
int CalibLoadFromFlash (TARGET_CONDI *cond)
{
    uint32 buf [8];
    flash_read_page (0, CALIB_FLASH_PAGE_TGT, buf, 8);
    if (buf[0] != CALIB_MAGIC) return 0;

    saved_tgt_rgb[0].red   = (unsigned char)( buf[1]        & 0xFF);
    saved_tgt_rgb[0].green = (unsigned char)((buf[1] >> 8)  & 0xFF);
    saved_tgt_rgb[0].blue  = (unsigned char)((buf[1] >> 16) & 0xFF);
    saved_tgt_H_min[0] = (unsigned char)((buf[1] >> 24) & 0xFF);
    saved_tgt_H_max[0] = (unsigned char)( buf[2]        & 0xFF);
    saved_tgt_S_min[0] = (unsigned char)((buf[2] >> 8)  & 0xFF);
    saved_tgt_S_max[0] = (unsigned char)((buf[2] >> 16) & 0xFF);
    saved_tgt_L_min[0] = (unsigned char)((buf[2] >> 24) & 0xFF);
    saved_tgt_L_max[0] = (unsigned char)(buf[3] & 0xFF);
    saved_tgt_valid[0] = 1;

    /* 同步到 cond 和当前工作变量 */
    cond->H_MIN = saved_tgt_H_min [0];  cond->H_MAX = saved_tgt_H_max [0];
    cond->S_MIN = saved_tgt_S_min [0];  cond->S_MAX = saved_tgt_S_max [0];
    cond->L_MIN = saved_tgt_L_min [0];  cond->L_MAX = saved_tgt_L_max [0];
    cond->WIDTH_MIN  = 3;   cond->HIGHT_MIN  = 3;
    cond->WIDTH_MAX  = 100; cond->HIGHT_MAX  = 100;

    calib_target  = saved_tgt_rgb[0];
    calib_H_min   = saved_tgt_H_min[0];  calib_H_max   = saved_tgt_H_max[0];
    calib_S_min   = saved_tgt_S_min[0];  calib_S_max   = saved_tgt_S_max[0];
    calib_L_min   = saved_tgt_L_min[0];  calib_L_max   = saved_tgt_L_max[0];
    calib_valid   = 1;
    return 1;
}

//==================== 9c. 保存参照色到 Flash ====================
/*
 * 将当前参照色（第 0 组）的简化 RGB 阈值写入 Flash 页 10。
 */
static void CalibSaveRefToFlash (void)
{
    uint32 buf [8];
    if (calib_round != 0) return;

    buf [0] = CALIB_MAGIC_REF;
    buf [1] = ((uint32) calib_ref_rgb.red)           |
              ((uint32) calib_ref_rgb.green    << 8)  |
              ((uint32) calib_ref_rgb.blue     << 16) |
              ((uint32) ref_rgb_range.R_MIN    << 24);
    buf [2] = ((uint32) ref_rgb_range.R_MAX)          |
              ((uint32) ref_rgb_range.G_MIN    << 8)   |
              ((uint32) ref_rgb_range.G_MAX    << 16)  |
              ((uint32) ref_rgb_range.B_MIN    << 24);
    buf [3] = ((uint32) ref_rgb_range.B_MAX)          |
              ((uint32) ref_dominant_channel   << 8);
    buf [4] = 0;   // 预留
    buf [5] = 0;
    buf [6] = 0;
    buf [7] = 0;

    flash_erase_page(0, CALIB_FLASH_PAGE_REF);
    flash_write_page(0, CALIB_FLASH_PAGE_REF, buf, 8);
}

//==================== 9d. 从 Flash 加载参照色 ====================
/*

 */

{
    uint32 buf [8];
    flash_read_page (0, CALIB_FLASH_PAGE_REF, buf, 8);
    if (buf[0] != CALIB_MAGIC_REF) return 0;

    saved_ref_rgb[0].red   = (unsigned char)( buf[1]        & 0xFF);
    saved_ref_rgb[0].green = (unsigned char)((buf[1] >> 8)  & 0xFF);
    saved_ref_rgb[0].blue  = (unsigned char)((buf[1] >> 16) & 0xFF);
    saved_ref_range[0].R_MIN = (unsigned char)((buf[1] >> 24) & 0xFF);
    saved_ref_range[0].R_MAX = (unsigned char)( buf[2]        & 0xFF);
    saved_ref_range[0].G_MIN = (unsigned char)((buf[2] >> 8)  & 0xFF);
    saved_ref_range[0].G_MAX = (unsigned char)((buf[2] >> 16) & 0xFF);
    saved_ref_range[0].B_MIN = (unsigned char)((buf[2] >> 24) & 0xFF);
    saved_ref_range[0].B_MAX  = (unsigned char)( buf[3]        & 0xFF);

//==================== 10. FindColorBlock：范式 A 双色追踪（多阈值版） ====================
/*
 * 核心识别函数，每帧调用一次。
 * 流程：
 *   1. 统计已校准阈值组数，若为 0 则返回失败。
 *   2. 使用第一组有效的参照色快速搜索（FindRefColorBlock），得到参照色包围盒，
 *      用于约束后续目标色搜索范围。
 *   3. 遍历所有已校准的目标色阈值组，对每组执行：
 *      a. 根据状态机（FULL_FRAME / LAST_FRAME / TRACING）搜索种子。
 *      b. 粗填充（宽松 HSL）获得区域平均 RGB。
 *      c. 计算平均 HSL，以此为中心收紧容差（精填充），得到精确边界。
 *      d. 空间验证（目标质心在参照色盒内）。
 *      e. 尺寸合法性检查。
 *      f. 任意一组通过则返回成功，并更新追踪状态。
 *   4. 若所有组都失败，返回 0。
 *
 * 参数：
 *   res      输出包围框（中心 x/y，宽高）
 *   avg_rgb  输出平均 RGB（用于显示）
 * 返回 1 找到目标，0 未找到。
 */
int FindColorBlock (RESULT *res, COLOR_RGB *avg_rgb)
{
    static TRACE_STATE  state = TS_FIND_IN_FULL_FRAME;
    static RESULT       last_res;
    static SEARCH_AREA  last_area;
    SEARCH_AREA search_area;
    const SEARCH_AREA full_area = {CAM_X_START, CAM_X_END, IMG_Y, IMG_Y + IMG_H};
    RESULT      pass1;
    COLOR_RGB   region_avg;
    unsigned long r_sum, g_sum, b_sum;
    unsigned int  p_cnt;
    COLOR_HSL    avg_hsl;
    unsigned char tH_min, tH_max, tS_min, tS_max, tL_min, tL_max;
    int          set_idx;
    int          total_sets;
    int          ref_found = 0;
    int          any_valid = 0;

    /* 统计已完成的校准组数（有目标色即可） */
    total_sets = 0;
    for (set_idx = 0; set_idx < NUM_THRESHOLD_SETS; set_idx++)
    {
        if (saved_tgt_valid [set_idx])
            total_sets++;
    }
    g_try_total = total_sets;
    g_try_current = 0;

    if (total_sets == 0) {
        res->x = res->y = res->w = res->h = 0;
        return 0;
    }

    /* ======== 阶段 0: 参照色快速搜索（每帧只跑一次，使用第一组有效 ref） ======== */
    {
        int first_ref_set = -1;
        for (set_idx = 0; set_idx < NUM_THRESHOLD_SETS; set_idx++)
        {
            if (saved_tgt_valid [set_idx] && saved_ref_valid [set_idx])
            {
                first_ref_set = set_idx;
                break;
            }
        }
        if (first_ref_set >= 0)
        {
            /* 临时切换到该组的参照色数据 */
            ref_rgb_range        = saved_ref_range [first_ref_set];
            ref_dominant_channel = saved_ref_dominant [first_ref_set];
            ref_found = FindRefColorBlock(&g_last_ref_res);
            if (ref_found)
            {
                /* 用参照色包围盒约束目标色搜索范围（加 margin） */
                search_area.X_Start = (g_last_ref_res.x > 10) ? (g_last_ref_res.x - 10) : CAM_X_START;
                search_area.Y_Start = (g_last_ref_res.y > 10) ? (g_last_ref_res.y - 10) : IMG_Y;
                search_area.X_End   = (g_last_ref_res.x + g_last_ref_res.w + 10 < CAM_X_END)
                                      ? (g_last_ref_res.x + g_last_ref_res.w + 10) : CAM_X_END;
                search_area.Y_End   = (g_last_ref_res.y + g_last_ref_res.h + 10 < IMG_Y + IMG_H)
                                      ? (g_last_ref_res.y + g_last_ref_res.h + 10) : IMG_Y + IMG_H;
            }
            else
            {
                /* 参照色丢失 → 降级全图搜索 */
                search_area = full_area;
            }
        }
        else
        {
            search_area = full_area;
        }
    }

    /* ======== 遍历所有有效阈值组 ======== */
    for (set_idx = 0; set_idx < NUM_THRESHOLD_SETS; set_idx++)
    {
        if (!saved_tgt_valid [set_idx])
            continue;

        g_try_current = set_idx + 1;   // 更新当前尝试序号

        /* 加载该组的阈值 */
        {
            unsigned char set_H_min = saved_tgt_H_min [set_idx];
            unsigned char set_H_max = saved_tgt_H_max [set_idx];
            unsigned char set_S_min = saved_tgt_S_min [set_idx];
            unsigned char set_S_max = saved_tgt_S_max [set_idx];
            unsigned char set_L_min = saved_tgt_L_min [set_idx];
            unsigned char set_L_max = saved_tgt_L_max [set_idx];
            unsigned int seed_x = 80, seed_y = 60;
            int seed_ok = 0;

            /* ---- 阶段 1: 寻找种子（状态机） ---- */
            switch (state)
            {
            case TS_TRACING:
                seed_x = last_res.x;
                seed_y = last_res.y;
                {
                    COLOR_RGB srgb;
                    COLOR_HSL shsl;
                    ReadColor(seed_x, seed_y, &srgb);
                    RGB888_to_HSL(&srgb, &shsl);
                    if (ColorMatchHSL(&shsl, set_H_min, set_H_max,
                                      set_S_min, set_S_max,
                                      set_L_min, set_L_max))
                    {
                        seed_ok = 1;
                        break;
                    }
                }
                state = TS_FIND_IN_LAST_FRAME;
                /* FALLTHROUGH */
            case TS_FIND_IN_LAST_FRAME:
                if (GridSearchSeed(&seed_x, &seed_y, set_H_min, set_H_max,
                                   set_S_min, set_S_max, set_L_min, set_L_max,
                                   3, &last_area))
                {
                    seed_ok = 1;
                    state = TS_TRACING;
                    break;
                }
                state = TS_FIND_IN_FULL_FRAME;
                /* FALLTHROUGH */
            case TS_FIND_IN_FULL_FRAME:
                if (GridSearchSeed (&seed_x, &seed_y, set_H_min, set_H_max,
                                   set_S_min, set_S_max, set_L_min, set_L_max,
                                   2, &search_area))
                {
                    seed_ok = 1;
                    state = TS_TRACING;
                    break;
                }
                /* 该组在全图范围也没找到种子，继续尝试下一组 */
                continue;
            }

            if (!seed_ok) continue;

            /* ---- 阶段 2: 粗填充（宽松阈值） ---- */
            if (!FloodFillHSL(seed_x, seed_y,
                              set_H_min, set_H_max,
                              set_S_min, set_S_max,
                              set_L_min, set_L_max,
                              MAX_FILL_PIXELS, &pass1,
                              &r_sum, &g_sum, &b_sum, &p_cnt))
            {
                state = TS_FIND_IN_LAST_FRAME;
                continue;  // 粗填充失败，尝试下一组
            }

            /* 过滤极小噪声区域 */
            if (p_cnt < 4)
            {
                state = TS_FIND_IN_LAST_FRAME;
                continue;
            }

            /* ---- 计算区域平均 RGB --> HSL ---- */
            region_avg.red   = (unsigned char)(r_sum / p_cnt);
            region_avg.green = (unsigned char)(g_sum / p_cnt);
            region_avg.blue  = (unsigned char)(b_sum / p_cnt);
            RGB888_to_HSL (&region_avg, &avg_hsl);

            /* H/S 保持与粗填充一致，L 收紧（TIGHT 容差） */
            tH_min = (unsigned char)((avg_hsl.hue > TIGHT_H_TOL) ? (avg_hsl.hue - TIGHT_H_TOL) : 0);
            tH_max = (unsigned char)((avg_hsl.hue + TIGHT_H_TOL < 240) ? (avg_hsl.hue + TIGHT_H_TOL) : 240);
            tS_min = (unsigned char)((avg_hsl.saturation > TIGHT_S_TOL) ? (avg_hsl.saturation - TIGHT_S_TOL) : 0);
            tS_max = (unsigned char)((avg_hsl.saturation + TIGHT_S_TOL < 240) ? (avg_hsl.saturation + TIGHT_S_TOL) : 240);
            tL_min = (unsigned char)((avg_hsl.luminance > TIGHT_L_TOL) ? (avg_hsl.luminance - TIGHT_L_TOL) : 0);
            tL_max = (unsigned char)((avg_hsl.luminance + TIGHT_L_TOL < 240) ? (avg_hsl.luminance + TIGHT_L_TOL) : 240);

            /* ---- 阶段 3: 精填充 ---- */
            if (!FloodFillHSL (pass1.x, pass1.y,
                              tH_min, tH_max, tS_min, tS_max, tL_min, tL_max,
                              MAX_FILL_PIXELS, res,
                              NULL, NULL, NULL, NULL))
            {
                state = TS_FIND_IN_LAST_FRAME;
                continue;
            }

            /* ---- 阶段 3.5: 空间验证（范式 A） ---- */
            if (ref_found && !SpatialVerify(&g_last_ref_res, res))
            {
                state = TS_FIND_IN_LAST_FRAME;
                continue;
            }

            /* ---- 验证尺寸 ---- */
            if (res->w < 2 || res->h < 2 || res->w > 160 || res->h > 120)
            {
                state = TS_FIND_IN_LAST_FRAME;
                continue;
            }

            /* ---- 输出平均色 ---- */
            if (avg_rgb) *avg_rgb = region_avg;

            /* ---- 更新追踪状态 ---- */
            last_res = *res;
            {
                unsigned int lw = res->w;
                unsigned int lh = res->h;
                last_area.X_Start = (res->x > lw) ? (res->x - lw) : CAM_X_START;
                last_area.X_End   = (res->x + lw < CAM_X_END) ? (res->x + lw) : CAM_X_END;
                last_area.Y_Start = (res->y > lh) ? (res->y - lh) : IMG_Y;
                last_area.Y_End   = (res->y + lh < IMG_Y + IMG_H) ? (res->y + lh) : (IMG_Y + IMG_H);
            }
            state = TS_TRACING;
            return 1;   // 找到目标
        }
    }

    /* 所有阈值组均失败 */
    res->x = res->y = res->w = res->h = 0;
    return 0;
}

//==================== 11. RGB565 → 亮度（BT.601 整数近似） ====================
/*
 * 计算 16 位 RGB565 像素的亮度 Y（0~255）。
 * 公式：Y = 0.299R + 0.587G + 0.114B，近似为 (R*77 + G*150 + B*29) >> 8。
 * 用于自动曝光统计。
 */
static uint8 rgb565_to_luminance(uint16 rgb565)
{
    uint8 r, g, b;
#if SCC8660_SWAP_BYTES
    rgb565 = ((rgb565 & 0xFF) << 8) | ((rgb565 >> 8) & 0xFF);
#endif
    r = (uint8)(((rgb565 >> 11) & 0x1F) * 255 / 31);
    g = (uint8)(((rgb565 >> 5)  & 0x3F) * 255 / 63);
    b = (uint8)(( rgb565        & 0x1F) * 255 / 31);
    return (uint8)(((uint16)r * 77 + (uint16)g * 150 + (uint16)b * 29) >> 8);
}

//==================== 12. ColorTrackingProcess：单帧完整流水线 ====================
/*
 * 主处理函数，每帧调用一次，完成以下步骤：
 *   1. 显示摄像头画面（160x120，左上角）
 *   2. 检测校准按键，若按下则调用 CalibrateFromCenter 进行校准
 *   3. 若从未校准，尝试从 Flash 加载（第 0 组）
 *   4. 调用 FindColorBlock 进行色块追踪
 *   5. 若找到，绘制包围框；若丢失，按 LOST_HOLD_FRAMES 保持坐标或归中
 *   6. 显示调试信息（校准状态、尝试组数等）
 *
 * 参数：
 *   cond     目标阈值结构（屏幕显示用）
 *   result   输出结果（中心坐标 + 尺寸）
 *   avg_rgb  输出平均 RGB
 * 返回 1 表示当前帧找到了目标（可能为保持坐标），0 表示未找到且归中。
 */
int ColorTrackingProcess (TARGET_CONDI *cond, RESULT *result, COLOR_RGB *avg_rgb)
{

    int         found;
    COLOR_RGB   Rgb;

//    auto_exposure_adjust_color (scc8660_image, 110);

    if (gpio_get_level(CALIB_BUTTON) == 0)
    {
        CalibrateFromCenter(cond);
        if (calib_state == CALIB_STATE_DONE)
            calib_done = 1;
        system_delay_ms (200);
        while (gpio_get_level (CALIB_BUTTON) == 0);
        system_delay_ms (5+0);
    }

    /* ---- 如果从未校准，尝试从 Flash 加载 ---- */
    if (!calib_done)
    {
        int ref_loaded = CalibLoadRefFromFlash ();   // 加载参照色（页 10）
        int tgt_loaded = CalibLoadFromFlash (cond);   // 加载目标色（页 11）
        if (ref_loaded && tgt_loaded)
        {
            calib_round = 1;                    // 已有 1 组，下一轮从索引 1 开始
            calib_state = CALIB_STATE_DONE;
            calib_done = 1;
        }
        else if (tgt_loaded)
        {
            // 只有目标色数据 → 降级为单色模式（无参照色约束）
            calib_round = 1;
            calib_state = CALIB_STATE_DONE;
            calib_ref_valid = 0;
            calib_done = 1;
        }
        else if (ref_loaded)
        {
            // 只有参照色 → 等待校准目标色
            calib_round = 0;
            calib_state = CALIB_STATE_REF_DONE;
        }
    }

    /* ---- 3. 色块搜索（含丢帧坐标保持） ---- */
    {
        static int    lost_frames = 0;
        static RESULT memory_res;
        if (calib_done)
            found = FindColorBlock(result, avg_rgb);
        else
            found = 0;

        if (found)
        {
            lost_frames = 0;
            memory_res  = *result;   // 保存当前结果
        }
        else
        {
            lost_frames++;
            if (lost_frames <= LOST_HOLD_FRAMES && memory_res.w > 0)
            {
                // 短暂丢失：保持上一帧坐标
                *result = memory_res;
                found   = 1;
            }
            else
            {
                // 连续丢失超过阈值：x 归中，其余清零
                result->x = 80;
                result->y = result->w = result->h = 0;
            }
        }
    }

    /* ---- 4. 绘制包围框 ---- */
    if (found)
    {
        uint16 half_w = (uint16)(result->w / 2);
        uint16 half_h = (uint16)(result->h / 2);
        uint16 top    = (result->y > half_h) ? (uint16)(result->y - half_h) : 0;
        uint16 bottom = (result->y + half_h < ips200_height_max)
                        ? (uint16)(result->y + half_h) : (ips200_height_max - 1);
        uint16 left   = (result->x > half_w) ? (uint16)(result->x - half_w) : 0;
        uint16 right  = (result->x + half_w < ips200_width_max)
                        ? (uint16)(result->x + half_w) : (ips200_width_max - 1);
        ips200_draw_line(left,  top,    right, top,    RGB565_GREEN);
        ips200_draw_line(left,  bottom, right, bottom, RGB565_GREEN);
        ips200_draw_line(left,  top,    left,  bottom, RGB565_GREEN);
        ips200_draw_line(right, top,    right, bottom, RGB565_GREEN);
    }

    /* ---- 5. 校准状态提示（多轮次） ---- */
    if (!calib_done)
    {
        if (calib_state == CALIB_STATE_NONE)
        {
            ips200_show_string(0, 167,  "P11_2:Calib REF");
            ips200_show_uint (120, 167, calib_round + 1, 1);
            ips200_show_string(130, 167, "/");
            ips200_show_uint (140, 167, NUM_THRESHOLD_SETS, 1);
        }
        else if (calib_state == CALIB_STATE_REF_DONE)
        {
            ips200_show_string(0, 167,  "P11_2:Calib TGT");
            ips200_show_uint (120, 167, calib_round + 1, 1);
            ips200_show_string(130, 167, "/");
            ips200_show_uint (140, 167, NUM_THRESHOLD_SETS, 1);
        }
    }

    /* ---- 5b. 多阈值尝试信息（追踪时显示） ---- */
    if (calib_done)
    {
        ips200_show_string(0, 167, "Try:");
        ips200_show_uint (40, 167, g_try_current, 1);
        ips200_show_string(50, 167, "/");
        ips200_show_uint (60, 167, g_try_total, 1);
    }

    // 以下为详细调试信息（注释掉，按需启用）
    // ...

    return found;
}

通过与华南赛区省赛的其他参赛人员交流后我们得知了一种叫斯坦利的算法，斯坦利算法是一种控制卡丁车转向的算法，它的原理很简单，在我们原有的给转向提供理论角度与实际角度的误差的基础之上，再加入一个以小车行驶中横向误差乘以一个系数k作为分子，以小车的实际速度作为分母作为自变量的反正切函数，它就是利用了反正切函数的单调性，在速度保持不变的情况下假如横向误差正向变大，那么反三角函数返回的值也就会随着增大，从而达到控制转向的效果，此算法就可以使小车在直线运行中不断的微操调整自身的姿态
下面是具体代码实现
/*斯坦利角度修正，放角度PID目标角度里*/
//要注意，由于是倒车，所以角度角度可能要反转180°
//A:上一个跟踪的点（可以理解是向量的起点），B:下一个要到达的点（当前目标终点），P:现在小车实际的坐标点
float stanly(float angle,float deg,float speed,float k,float ks,float A_x,float A_y,float B_x,float B_y)
{
    float len,px_mj,AB_x,AB_y,AP_x,AP_y,hx_err;

    qj.stanley_heading_error_deg=deg-angle;//角度误差=理想角度-实际角度

    if(qj.stanley_heading_error_deg>180.0f){//角度归一化
        qj.stanley_heading_error_deg-=360.0f;
    }else if(qj.stanley_heading_error_deg<-180.0f){
        qj.stanley_heading_error_deg+=360.0f;
    }

    AB_x=B_x-A_x;//向量计算
    AB_y=B_y-A_y;
    AP_x=my_now_x-A_x;
    AP_y=my_now_y-A_y;

    len=sqrtf(AB_x*AB_x+AB_y*AB_y);//当前跟踪路径点与下一个点的直线距离（平行四边形底边）

    px_mj=AB_x*AP_y-AB_y*AP_x;//向量叉乘（平行四边形面积）

    if(len<0.001f){//防止除0
        len=0.001f;
    }

    qj.stanley_cross_track_error=px_mj/len;//求出高（横向误差）

    qj.stanley_cross_track_error=-qj.stanley_cross_track_error;

    qj.stanley_speed_mps=speed*10;//小车速度

    hx_err=atanf((k*qj.stanley_cross_track_error)/(qj.stanley_speed_mps+ks))*180.0f/DR_PI_F;//三角函数项

   // hx_err=0;

    qj.stanley_steer_raw_deg=qj.stanley_heading_error_deg+hx_err;//最终公式=角度误差+三角函数项

    if(qj.stanley_steer_raw_deg>35){//限幅-35°~35°
        qj.stanley_steer_raw_deg=35;
    }else if(qj.stanley_steer_raw_deg<-35){
        qj.stanley_steer_raw_deg=-35;
    }

    return qj.stanley_steer_raw_deg;
}

    now_cont=cont;

        last_cont=0;
    }*/

    cont_dt=now_cont-last_cont;

    last_cont=now_cont;

    a_x=cont_dt*sin(ange*DR_PI_F/180)/DW;//求微分(变化量)
    a_y=cont_dt*cos(ange*DR_PI_F/180)/DW;

    my_now_x+=a_x;//进行积分
    my_now_y+=a_y;



}

//实时计算角度还有两点间距离
//灵鹿工作室羁绊
//参数：需要追踪的点位，当前点位，需要赋值的角度、两点间距离变量地址
void Linglu_GZS_JB(float next_x,float next_y,float now_x,float now_y,float *angle,float *distance)
{
   float dx,dy,my_anger,my_distance;

   dx=next_x-now_x;
   dy=next_y-now_y;

   my_anger=atan2f(dx,dy)*180.0f/DR_PI_F;//计算两点间角度

   if(my_anger<0){
    my_anger+=360.0f;//atan2f值域是-180~180,对负数区域进行+=360°处理
   }
   if(my_anger>=180){//角度反转，得到小车正向角度
       my_anger-=180;
   }else if(my_anger<180){
       my_anger+=180;
   }

   my_distance=sqrtf(dx*dx+dy*dy);

   *angle=my_anger;
   *distance=my_distance;


}

所有的跟wifi串口有关的代码都已改成符合wifi-spi模块的逻辑，并且程序也无报错，可是似乎通过逐飞助手上的信息观察并没有执行语音识别的任务，我想到用printf函数测试程序运行是否正常逐步排查，最终发现原来是获取时间戳的环节有问题，printf函数发送到逐飞助手上显示到科大讯飞服务器返回的时间戳信息是空的，我推测要么是发送环节有问题，要么是接受环节有问题，甚至是两个都有问题，于是我对发送和接收函数使用了while循环，确保发送能发送完，接收能接收到，结果显示成功获取到时间戳信息，然而语音识别功能依然无法使用，我感受到这是个很大的挑战

日志26.3.10
我利用昨天用printf函数一步补排查检查程序是否按顺序正常运行以及显示一些重要的数据是否正常显示，一步步找出问题，发现原来是代码里面有一个字符串信息查找多了一个空格，导致程序一直无法匹配到正确的信息，所以导致程序无法与云端建立通信协议，删除以后可以正确读取正确的信息

日志26.3.15
现在基本实现WIFI模块与云端建立wedsocket协议，我用逐飞助手连接上串口在逐飞助手上面显示云端发送过来的信息，结果是显示了一大串乱码，我无论是用UTF8还是HEX都无法正常接收汉字，我开始手足无措，因为针对这种现象我无从下手

日志26.3.20
今天终于在偶然间发现显示乱码问题的根源所在，我注意到在原本的例程里面，在WIFI模块发送和接收函数内存在大量的delay延时函数，这些delay的本意是为了保证WIFI串口能够发送和接收完整的数据，但是我们使用的是WIFI-SPI模块其响应速度更快，这些delay函数反而拖累了WIFI-SPI的响应，所以麦克风采集的数据其实是没问题的，但是由于delay的影响，使SPI不能来得及接收数据（个人推测），所以导致了接收的数据显示出来的是乱码，我减少了这些延时，成功实现语音识别，接下来就是写具体的任务执行代码了
